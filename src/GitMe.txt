# ── HELPER: safe null check (PS 5.1 compatible, no ??) ────────────────────────
function Coalesce($a, $b) { if ($null -ne $a -and $a -ne "") { $a } else { $b } }

# ── PULL GROUPS ───────────────────────────────────────────────────────────────
Write-Host "[3/5] Pulling security groups and members..." -ForegroundColor Cyan

$allGroups = Get-MgGroup -Filter "securityEnabled eq true" -All `
             -Property "DisplayName,Id,Description" |
             Where-Object {
                 $n = $_.DisplayName
                 $match = $false
                 foreach ($p in $GroupPrefixes) {
                     if ($n -like "$p*") { $match = $true }
                 }
                 $match
             }

Write-Host "      Found $($allGroups.Count) groups" -ForegroundColor Green

$roleAssignments = Get-AzRoleAssignment |
                   Where-Object { $_.ObjectType -eq "Group" } |
                   Select-Object DisplayName, RoleDefinitionName, Scope

$groupsData = New-Object System.Collections.ArrayList

foreach ($g in $allGroups) {
    Write-Host "      Processing: $($g.DisplayName)" -ForegroundColor Gray

    # Find role assignment for this group
    $role = $roleAssignments | Where-Object { $_.DisplayName -eq $g.DisplayName } | Select-Object -First 1
    if ($role) {
        $roleName  = $role.RoleDefinitionName
        $roleScope = $role.Scope
    } else {
        $roleName  = "No direct assignment"
        $roleScope = ""
    }

    # Get group description safely
    $groupDesc = Coalesce $g.Description ""

    # Get members
    $rawMembers = Get-MgGroupMember -GroupId $g.Id -All -ErrorAction SilentlyContinue
    $membersList = New-Object System.Collections.ArrayList

    foreach ($m in $rawMembers) {
        $user = Get-MgUser -UserId $m.Id -Property "UserPrincipalName,DisplayName" -ErrorAction SilentlyContinue
        if ($user) {
            # PS 5.1 safe: get added date without ??
            $addedDate = "Unknown"
            if ($m.AdditionalProperties -and $m.AdditionalProperties.ContainsKey("createdDateTime")) {
                $addedDate = $m.AdditionalProperties["createdDateTime"]
            }
            $memberObj = @{
                upn   = $user.UserPrincipalName
                name  = $user.DisplayName
                type  = "User"
                added = $addedDate
            }
            [void]$membersList.Add($memberObj)
        } else {
            $sp = Get-MgServicePrincipal -ServicePrincipalId $m.Id -ErrorAction SilentlyContinue
            if ($sp) {
                $memberObj = @{
                    upn   = Coalesce $sp.AppId $sp.DisplayName
                    name  = $sp.DisplayName
                    type  = "Service Principal"
                    added = "Unknown"
                }
                [void]$membersList.Add($memberObj)
            }
        }
    }

    $groupObj = @{
        name    = $g.DisplayName
        desc    = $groupDesc
        role    = $roleName
        scope   = $roleScope
        sub     = $SubName
        members = $membersList.ToArray()
    }
    [void]$groupsData.Add($groupObj)
}

# Helper functions — PS 5.1 compatible
function Resolve-Principal($id) {
    $u = Get-MgUser -UserId $id -ErrorAction SilentlyContinue
    if ($u) { return $u.UserPrincipalName }
    $sp = Get-MgServicePrincipal -ServicePrincipalId $id -ErrorAction SilentlyContinue
    if ($sp) { return $sp.DisplayName }
    return $id
}

function Resolve-Role($id) {
    $r = Get-MgRoleManagementDirectoryRoleDefinition -UnifiedRoleDefinitionId $id -ErrorAction SilentlyContinue
    if ($r) { return $r.DisplayName } else { return $id }
}

function Get-Tier($roleName) {
    if ($roleName -match "Owner|Global Admin|Privileged") { return "Critical" }
    if ($roleName -match "Contributor|User Access")       { return "High" }
    if ($roleName -match "Reader|Monitor")                { return "Low" }
    return "Medium"
}

# Active PIM assignments
$pimActiveData = New-Object System.Collections.ArrayList
try {
    $active = Get-MgRoleManagementDirectoryRoleAssignmentScheduleInstance -All -ErrorAction Stop |
              Where-Object { $_.AssignmentType -eq "Activated" }
    foreach ($a in $active) {
        $startStr = if ($a.StartDateTime) { $a.StartDateTime.ToString("yyyy-MM-ddTHH:mm:ssZ") } else { "" }
        $endStr   = if ($a.EndDateTime)   { $a.EndDateTime.ToString("yyyy-MM-ddTHH:mm:ssZ") }   else { "" }
        $dur      = "Unknown"
        if ($a.StartDateTime -and $a.EndDateTime) {
            $dur = "$([Math]::Round(($a.EndDateTime - $a.StartDateTime).TotalHours, 1))h"
        }
        $pimObj = @{
            principal   = Resolve-Principal $a.PrincipalId
            role        = Resolve-Role $a.RoleDefinitionId
            scope       = Coalesce $a.DirectoryScopeId "/"
            activatedAt = $startStr
            expiresAt   = $endStr
            duration    = $dur
        }
        [void]$pimActiveData.Add($pimObj)
    }
} catch {
    Write-Host "      PIM active assignments: $($_.Exception.Message)" -ForegroundColor Yellow
}

# Eligible (JIT) assignments
$pimEligData = New-Object System.Collections.ArrayList
try {
    $elig = Get-MgRoleManagementDirectoryRoleEligibilitySchedule -All -ErrorAction Stop
    foreach ($e in $elig) {
        $rName = Resolve-Role $e.RoleDefinitionId
        $maxDur = "Not set"
        if ($e.ScheduleInfo -and $e.ScheduleInfo.Expiration -and $e.ScheduleInfo.Expiration.Duration) {
            $maxDur = $e.ScheduleInfo.Expiration.Duration
        }
        $pimObj = @{
            principal   = Resolve-Principal $e.PrincipalId
            role        = $rName
            tier        = Get-Tier $rName
            maxDuration = $maxDur
            approval    = "Check PIM policy"
            mfa         = "Check PIM policy"
        }
        [void]$pimEligData.Add($pimObj)
    }
} catch {
    Write-Host "      PIM eligible: $($_.Exception.Message)" -ForegroundColor Yellow
}

# PIM activation history — last 30 days
$pimHistData = New-Object System.Collections.ArrayList
try {
    $since  = (Get-Date).AddDays(-30).ToString("yyyy-MM-ddTHH:mm:ssZ")
    $audits = Get-MgAuditLogDirectoryAudit -All -ErrorAction Stop |
              Where-Object { $_.LoggedByService -eq "PIM" -and $_.ActivityDateTime -ge [datetime]$since }
    foreach ($a in $audits) {
        # Safely extract fields
        $principal = "Unknown"
        if ($a.InitiatedBy.User -and $a.InitiatedBy.User.UserPrincipalName) {
            $principal = $a.InitiatedBy.User.UserPrincipalName
        } elseif ($a.InitiatedBy.App -and $a.InitiatedBy.App.DisplayName) {
            $principal = $a.InitiatedBy.App.DisplayName
        }

        $role = "Unknown"
        $roleTarget = $a.TargetResources | Where-Object { $_.Type -eq "Role" } | Select-Object -First 1
        if ($roleTarget) { $role = $roleTarget.DisplayName }

        $just = ""
        $approver = "N/A"
        if ($a.AdditionalDetails) {
            $justEntry     = $a.AdditionalDetails | Where-Object { $_.Key -eq "justification" } | Select-Object -First 1
            $approverEntry = $a.AdditionalDetails | Where-Object { $_.Key -eq "approver" }      | Select-Object -First 1
            if ($justEntry)     { $just     = $justEntry.Value }
            if ($approverEntry) { $approver = $approverEntry.Value }
        }

        $histObj = @{
            date          = $a.ActivityDateTime.ToString("yyyy-MM-dd HH:mm")
            principal     = $principal
            role          = $role
            justification = $just
            approvedBy    = $approver
            result        = $a.Result
        }
        [void]$pimHistData.Add($histObj)
    }
} catch {
    Write-Host "      PIM history: $($_.Exception.Message)" -ForegroundColor Yellow
}

Write-Host "      PIM active=$($pimActiveData.Count) eligible=$($pimEligData.Count) history=$($pimHistData.Count)" -ForegroundColor Green

function TierBadge($t) {
    $cls = switch ($t) { "Critical"{"b-crit"} "High"{"b-high"} "Low"{"b-low"} default{"b-med"} }
    "<span class='badge $cls'>$t</span>"
}

        "<td style='font-size:11px'>$just</td>" +
        "<td style='font-size:11px'>$($r.approvedBy)</td><td>$resBadge</td></tr>"
}

<script>
var GROUPS   = $groupsJson;
var PIM_HIST = $pimHistJson;
var META     = { date:"$ReviewDate", reviewer:"$ReviewedBy", sub:"$SubName" };
var dec = {};
var exp = {};

function showTab(id, el) {
  document.querySelectorAll('.tab-body').forEach(function(t){ t.classList.remove('active'); });
  document.querySelectorAll('.tab').forEach(function(t){ t.classList.remove('active'); });
  document.getElementById('tab-'+id).classList.add('active');
  el.classList.add('active');
  if (id === 'signoff') buildSummary();
}
function gd(id,i){ return dec[id+'_'+i] || ''; }
function sd(id,i,v){
  var k = id+'_'+i;
  dec[k] = (dec[k] === v) ? '' : v;
  sync(); render(); exp[id] = true; render();
}
function gStatus(g){
  var ds = g.members.map(function(_,i){ return gd(g.name,i); });
  if (ds.every(function(d){ return d===''; })) return 'pending';
  if (ds.some(function(d){ return d==='remove'; })) return 'action';
  if (ds.every(function(d){ return d!==''; })) return 'done';
  return 'pending';
}
function sync(){
  var rem=0, decided=0, grpDone=0;
  GROUPS.forEach(function(g){
    g.members.forEach(function(_,i){ if(gd(g.name,i)==='remove') rem++; if(gd(g.name,i)!=='') decided++; });
    if(g.members.every(function(_,i){ return gd(g.name,i)!==''; })) grpDone++;
  });
  var total = GROUPS.reduce(function(a,g){ return a+g.members.length; }, 0);
  document.getElementById('s-remove').textContent = rem;
  document.getElementById('sb-remove').textContent = rem;
  document.getElementById('sb-done').textContent = grpDone;
  document.getElementById('s-pct').textContent = Math.round(decided/Math.max(total,1)*100)+'%';
  var allDone = GROUPS.every(function(g){ return g.members.every(function(_,i){ return gd(g.name,i)!==''; }); });
  var os = document.getElementById('overall-status');
  os.textContent = allDone ? 'Complete — ready for sign-off' : 'In progress';
  os.style.color = allDone ? '#2e7d32' : '#e65100';
}
function render(){
  var q  = document.getElementById('search').value.toLowerCase();
  var fr = document.getElementById('f-role').value;
  var fs = document.getElementById('f-status').value;
  var filtered = GROUPS.filter(function(g){
    return (!q || g.name.toLowerCase().indexOf(q) > -1) &&
           (!fr || g.role === fr) &&
           (!fs || gStatus(g) === fs);
  });
  document.getElementById('cnt').textContent = 'Showing '+filtered.length+' of '+GROUPS.length+' groups';
  var html = '';
  filtered.forEach(function(g){
    var open = exp[g.name] ? 'open' : '';
    var st = gStatus(g);
    var sbadge = st==='done' ? "<span class='badge b-done'>Reviewed</span>" :
                 st==='action' ? "<span class='badge b-act'>Action required</span>" :
                 "<span class='badge b-pend'>Pending</span>";
    var ds = g.members.map(function(_,i){ return gd(g.name,i); });
    var done = ds.filter(function(d){ return d!==''; }).length;
    var pct  = g.members.length ? Math.round(done/g.members.length*100) : 0;
    var rem  = ds.filter(function(d){ return d==='remove'; }).length;
    var rows = '';
    g.members.forEach(function(m,i){
      var d = gd(g.name,i);
      var rc = d==='keep'?'rk':d==='remove'?'rr':'';
      var gn = g.name.replace(/'/g,"\\'");
      rows += "<tr class='"+rc+"'><td>"+m.upn+"</td><td>"+m.name+"</td><td>"+m.type+"</td><td>"+m.added+"</td>"+
        "<td><div class='dec-wrap'>"+
        "<button class='dbtn "+(d==='keep'?'dk':'')+"' onclick=\"sd('"+gn+"',"+i+",'keep')\">Keep</button>"+
        "<button class='dbtn "+(d==='remove'?'dr':'')+"' onclick=\"sd('"+gn+"',"+i+",'remove')\">Remove</button>"+
        "</div></td></tr>";
    });
    var scopeTxt = g.scope ? " <span style='color:#aaa;font-size:10px'>"+g.scope+"</span>" : "";
    html += "<div class='group'>"+
      "<div class='ghdr' onclick='tog(\""+g.name.replace(/"/g,'&quot;')+"\")'>"+
      "<span class='chev "+open+"'>&#9654;</span>"+
      "<span class='gname' title='"+g.name+"'>"+g.name+"</span>"+
      "<span class='badge b-sub'>"+g.sub+"</span>"+
      (g.role?"<span class='badge b-role'>"+g.role+"</span>":"")+
      "<span class='badge b-count'>"+g.members.length+" members</span>"+sbadge+"</div>"+
      "<div class='desc'>"+g.desc+scopeTxt+"</div>"+
      "<div class='panel "+open+"'>"+
      "<table><thead><tr><th>UPN</th><th>Display name</th><th>Type</th><th>Date added</th><th>Decision</th></tr></thead>"+
      "<tbody>"+rows+"</tbody></table>"+
      "<div class='footer-row'><span>"+done+"/"+g.members.length+" decided"+
      (rem?" &nbsp;|&nbsp; <strong style='color:#b71c1c'>"+rem+" flagged for removal</strong>":"")+"</span></div>"+
      "<div class='prog'><div class='prog-fill' style='width:"+pct+"%'></div></div>"+
      "</div></div>";
  });
  document.getElementById('list').innerHTML = html;

  // Populate role filter once
  var sel = document.getElementById('f-role');
  if (sel.options.length === 1) {
    var roles = [];
    GROUPS.forEach(function(g){ if(g.role && roles.indexOf(g.role)===-1) roles.push(g.role); });
    roles.sort().forEach(function(r){ var o=document.createElement('option'); o.value=r; o.textContent=r; sel.appendChild(o); });
  }
}
function tog(id){ exp[id] = !exp[id]; render(); }
function expandAll(){ GROUPS.forEach(function(g){ exp[g.name]=true; }); render(); }
function collapseAll(){ exp={}; render(); }
function filterPim(){
  var q  = document.getElementById('pim-search').value.toLowerCase();
  var r  = document.getElementById('pim-result').value;
  document.querySelectorAll('#pim-hist-body tr').forEach(function(tr){
    var txt = tr.textContent.toLowerCase();
    var res = tr.getAttribute('data-result') || '';
    tr.style.display = (!q||txt.indexOf(q)>-1)&&(!r||res===r) ? '' : 'none';
  });
}
function buildSummary(){
  var rows = '';
  GROUPS.forEach(function(g){
    g.members.forEach(function(m,i){
      var d = gd(g.name,i) || 'pending';
      var bg = d==='keep'?'#f1f8f1':d==='remove'?'#fff5f5':'';
      var fc = d==='remove'?'#b71c1c':d==='keep'?'#2e7d32':'#888';
      rows += "<tr style='background:"+bg+"'><td>"+g.name+"</td><td>"+m.upn+"</td><td>"+m.type+"</td>"+
        "<td><strong style='color:"+fc+"'>"+d+"</strong></td><td>"+g.role+"</td></tr>";
    });
  });
  document.getElementById('summary-body').innerHTML = rows;
}
function exportCSV(){
  var rows = ['Group,Subscription,Azure Role,Scope,UPN,Display Name,Type,Date Added,Decision,Review Date,Reviewer'];
  var notes = document.getElementById('so-notes').value.replace(/"/g,'""');
  GROUPS.forEach(function(g){
    g.members.forEach(function(m,i){
      var d = gd(g.name,i) || 'pending';
      rows.push([g.name,g.sub,g.role,g.scope,m.upn,m.name,m.type,m.added,d,META.date,META.reviewer]
        .map(function(v){ return '"'+String(v||'').replace(/"/g,'""')+'"'; }).join(','));
    });
  });
  rows.push('','PIM ACTIVATION HISTORY','Date,Principal,Role,Justification,Approved By,Result');
  PIM_HIST.forEach(function(h){
    rows.push([h.date,h.principal,h.role,h.justification,h.approvedBy,h.result]
      .map(function(v){ return '"'+String(v||'').replace(/"/g,'""')+'"'; }).join(','));
  });
  var a = document.createElement('a');
  a.href = 'data:text/csv;charset=utf-8,'+encodeURIComponent(rows.join('\n'));
  a.download = 'RBAC_Review_'+META.date+'.csv';
  a.click();
}
function signOff(){
  var name = document.getElementById('so-name').value;
  var date = document.getElementById('so-date').value;
  var stamp = document.getElementById('signed-stamp');
  stamp.style.display = 'block';
  stamp.textContent = 'Signed off by '+name+' on '+date+' — ready for audit submission.';
  buildSummary();
}
sync(); render();
</script>
</body></html>

$html | Out-File -FilePath $OutputPath -Encoding UTF8
