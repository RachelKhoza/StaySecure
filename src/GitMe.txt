[CmdletBinding()]
param(
    # Excel input
    [Parameter(Mandatory = $true)]
    [string]$ExcelFilePath,

    # Column in Excel that contains the "Names"
    [Parameter(Mandatory = $false)]
    [string]$ColumnName = "Name",

    # Optional sheet name
    [Parameter(Mandatory = $false)]
    [string]$SheetName
)

Set-StrictMode -Version Latest
$ErrorActionPreference = "Stop"

# -----------------------------
# Hardcoded values (as requested)
# -----------------------------
$ResourceGroupName = "SharedRG"

$AppGatewayNames = @(
    "agw-shared-app-01",
    "agw-shared-app-02"
)

$OutputCsv = Join-Path -Path (Split-Path -Parent $ExcelFilePath) -ChildPath "AGWMatcheReport.csv"

# -----------------------------
# Helpers
# -----------------------------
function Ensure-Module {
    param([Parameter(Mandatory)][string]$Name)

    if (-not (Get-Module -ListAvailable -Name $Name)) {
        Write-Host "[!] Module '$Name' not found. Installing from PSGallery..." -ForegroundColor Yellow
        Install-Module -Name $Name -Scope CurrentUser -Force -ErrorAction Stop
    }

    Import-Module -Name $Name -ErrorAction Stop
}

function Load-ExcelNames {
    param(
        [Parameter(Mandatory)][string]$Path,
        [Parameter(Mandatory)][string]$Column,
        [string]$Sheet
    )

    if (-not (Test-Path $Path)) {
        throw "Excel file not found: $Path"
    }

    $importParams = @{ Path = $Path }
    if ($Sheet) { $importParams.WorksheetName = $Sheet }

    $excelData = Import-Excel @importParams

    if (-not $excelData -or $excelData.Count -eq 0) {
        throw "No data found in Excel file."
    }

    $availableColumns = $excelData[0].PSObject.Properties.Name
    if ($Column -notin $availableColumns) {
        throw "Column '$Column' not found. Available columns: $($availableColumns -join ', ')"
    }

    # Unique, non-empty entries
    return $excelData |
        Select-Object -ExpandProperty $Column |
        Where-Object { $_ -and $_.ToString().Trim() -ne "" } |
        ForEach-Object { $_.ToString().Trim() } |
        Select-Object -Unique
}

function Get-AGWNamesInventory {
    param(
        [Parameter(Mandatory)][string]$ResourceGroup,
        [Parameter(Mandatory)][string]$AgwName
    )

    $appGw = Get-AzApplicationGateway -ResourceGroupName $ResourceGroup -Name $AgwName -ErrorAction Stop

    $ruleNames = @()
    $listenerNames = @()

    if ($appGw.RequestRoutingRules) {
        $ruleNames = $appGw.RequestRoutingRules | ForEach-Object { $_.Name }
    }

    if ($appGw.HttpListeners) {
        $listenerNames = $appGw.HttpListeners | ForEach-Object { $_.Name }
    }

    # Combine into a single searchable set
    return ($ruleNames + $listenerNames) | Where-Object { $_ } | Select-Object -Unique
}

function Is-Match {
    param(
        [Parameter(Mandatory)][string]$ExcelName,
        [Parameter(Mandatory)][string[]]$InventoryNames
    )

    # --- Matching strategy (aligned with your current approach) ---
    # Use first token (first word) and do a contains match
    $token = ($ExcelName -split '\s+')[0]

    if (-not $token -or $token.Length -lt 1) { return $false }

    return [bool]($InventoryNames | Where-Object { $_ -like "*$token*" } | Select-Object -First 1)

    # If you ever want FULL string contains match instead, use this:
    # return [bool]($InventoryNames | Where-Object { $_ -like "*$ExcelName*" } | Select-Object -First 1)

    # If you want EXACT match instead, use this:
    # return $InventoryNames -contains $ExcelName
}

# -----------------------------
# Execution
# -----------------------------
Ensure-Module "ImportExcel"
Ensure-Module "Az.Network"

Write-Host "`n========================================" -ForegroundColor Cyan
Write-Host "AGW Rule/Listener Matcher (Read-Only)" -ForegroundColor Cyan
Write-Host "========================================`n" -ForegroundColor Cyan

# Load Excel Names
$excelNames = Load-ExcelNames -Path $ExcelFilePath -Column $ColumnName -Sheet $SheetName
Write-Host "[+] Loaded $($excelNames.Count) unique names from Excel column '$ColumnName'." -ForegroundColor Green

# Build report rows: one row per (AGW, ExcelName)
$report = New-Object System.Collections.Generic.List[object]

foreach ($agw in $AppGatewayNames) {
    Write-Host "`n[*] Scanning AGW: $agw" -ForegroundColor Yellow

    $inventory = Get-AGWNamesInventory -ResourceGroup $ResourceGroupName -AgwName $agw
    Write-Host "    Found $($inventory.Count) total Rule/Listener names in $agw." -ForegroundColor Gray

    foreach ($name in $excelNames) {
        $matched = Is-Match -ExcelName $name -InventoryNames $inventory

        $report.Add([PSCustomObject]@{
            AppGateway  = $agw
            ExcelName   = $name
            Matched     = $matched   # True/False as requested
        })
    }
}

# Export
$report | Export-Csv -Path $OutputCsv -NoTypeInformation -Encoding UTF8

# Summary
$total = $report.Count
$trueCount = ($report | Where-Object { $_.Matched -eq $true }).Count
$falseCount = $total - $trueCount

Write-Host "`n========== SUMMARY ==========" -ForegroundColor Cyan
Write-Host "AGWs scanned      : $($AppGatewayNames.Count)"
Write-Host "Excel names        : $($excelNames.Count)"
Write-Host "Rows written       : $total"
Write-Host "Matched (True)     : $trueCount" -ForegroundColor Green
Write-Host "Not matched (False): $falseCount" -ForegroundColor Red
Write-Host "`n[+] Exported report to: $OutputCsv" -ForegroundColor Green
Write-Host "=============================`n" -ForegroundColor Cyan
