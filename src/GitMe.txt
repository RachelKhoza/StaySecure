$GroupPrefixes  = @("DRG-", "NORA-", "MGMT-")
# ─────────────────────────────────────────────────────────────────────────────

$ReviewDate = Get-Date -Format "yyyy-MM-dd"
$OutputPath = Join-Path (Get-Location) "RBAC_Review_$ReviewDate.html"

# ── HELPER: PS 5.1 safe null check ────────────────────────────────────────────
function Coalesce($a, $b) { if ($null -ne $a -and $a -ne "") { $a } else { $b } }

# ── CONNECT ───────────────────────────────────────────────────────────────────
Write-Host "`n[1/4] Connecting to Azure..." -ForegroundColor Cyan
Import-Module Az.Accounts -Force -ErrorAction SilentlyContinue
Import-Module Az.Resources -Force -ErrorAction SilentlyContinue

# Disable WAM (Web Account Manager) — forces browser-based auth, avoids
# Azure.Identity version conflict errors. Persists across sessions.
try {
    Update-AzConfig -EnableLoginByWam $false -Scope CurrentUser -ErrorAction SilentlyContinue | Out-Null
    Write-Host "      WAM disabled, using browser auth" -ForegroundColor DarkGray
} catch {
    # Older Az versions don't have this config — ignore silently
}

try {
    $existingContext = Get-AzContext -ErrorAction SilentlyContinue
    if (-not $existingContext -or $existingContext.Subscription.Id -ne $SubscriptionId) {
        Connect-AzAccount -SubscriptionId $SubscriptionId -ErrorAction Stop | Out-Null
    }
    Set-AzContext -SubscriptionId $SubscriptionId -ErrorAction Stop | Out-Null
    $SubName = (Get-AzContext).Subscription.Name
    Write-Host "      OK - Connected to: $SubName" -ForegroundColor Green
} catch {
    Write-Host "      FAILED: $($_.Exception.Message)" -ForegroundColor Red
    exit 1
}

# ── PULL GROUPS via Az AD ─────────────────────────────────────────────────────
Write-Host "[2/4] Pulling security groups via Az.Resources..." -ForegroundColor Cyan

$allGroups = New-Object System.Collections.ArrayList
foreach ($prefix in $GroupPrefixes) {
    Write-Host "      Searching prefix: $prefix" -ForegroundColor Gray
    try {
        $found = Get-AzADGroup -DisplayNameStartsWith $prefix -ErrorAction SilentlyContinue
        foreach ($g in $found) { [void]$allGroups.Add($g) }
        Write-Host "        Found $($found.Count) groups" -ForegroundColor DarkGray
    } catch {
        Write-Host "        Error: $($_.Exception.Message)" -ForegroundColor Yellow
    }
}

Write-Host "      Total groups matching prefixes: $($allGroups.Count)" -ForegroundColor Green

# ── PULL ROLE ASSIGNMENTS ─────────────────────────────────────────────────────
Write-Host "[3/4] Pulling role assignments and members..." -ForegroundColor Cyan

$roleAssignments = Get-AzRoleAssignment | Where-Object { $_.ObjectType -eq "Group" }
Write-Host "      Found $($roleAssignments.Count) group role assignments in this subscription" -ForegroundColor Green

$groupsData = New-Object System.Collections.ArrayList

foreach ($g in $allGroups) {
    Write-Host "      Processing: $($g.DisplayName)" -ForegroundColor Gray

    # Find role assignment
    $role = $roleAssignments | Where-Object { $_.DisplayName -eq $g.DisplayName } | Select-Object -First 1
    if ($role) {
        $roleName  = $role.RoleDefinitionName
        $roleScope = $role.Scope
    } else {
        $roleName  = "No direct assignment in this sub"
        $roleScope = ""
    }

    $groupDesc = Coalesce $g.Description ""

    # Get members via Az
    $membersList = New-Object System.Collections.ArrayList
    try {
        $rawMembers = Get-AzADGroupMember -GroupObjectId $g.Id -ErrorAction SilentlyContinue
        foreach ($m in $rawMembers) {
            $upn  = ""
            $name = $m.DisplayName
            $type = "User"

            if ($m.UserPrincipalName) {
                $upn  = $m.UserPrincipalName
                $type = "User"
            } elseif ($m.AppId) {
                $upn  = $m.AppId
                $type = "Service Principal"
            } else {
                $upn  = $m.Id
                $type = if ($m.OdataType -like "*group*") { "Group" } else { "Unknown" }
            }

            $memberObj = @{
                upn   = $upn
                name  = $name
                type  = $type
                added = "Not available via Az"
            }
            [void]$membersList.Add($memberObj)
        }
    } catch {
        Write-Host "        Could not get members: $($_.Exception.Message)" -ForegroundColor Yellow
    }

    $groupObj = @{
        name    = $g.DisplayName
        desc    = $groupDesc
        role    = $roleName
        scope   = $roleScope
        sub     = $SubName
        members = $membersList.ToArray()
    }
    [void]$groupsData.Add($groupObj)
}

# ── BUILD HTML ────────────────────────────────────────────────────────────────
Write-Host "[4/4] Writing HTML report..." -ForegroundColor Cyan

# Convert to JSON
$groupsJson = $groupsData.ToArray() | ConvertTo-Json -Depth 6 -Compress
if ($groupsJson -eq "null" -or [string]::IsNullOrEmpty($groupsJson)) { $groupsJson = "[]" }
# Ensure single object becomes array
if ($groupsJson -notmatch "^\[") { $groupsJson = "[$groupsJson]" }

$totalMembers = 0
foreach ($g in $groupsData) { $totalMembers += $g.members.Count }
$groupCount = $groupsData.Count

# Truncate sub name for display
if ($SubName.Length -gt 12) {
    $SubNameShort = $SubName.Substring(0, 12)
} else {
    $SubNameShort = $SubName
}

<script>
var GROUPS = $groupsJson;
var META = { date:"$ReviewDate", reviewer:"$ReviewedBy", sub:"$SubName" };
var dec = {};
var exp = {};

function showTab(id, el) {
  var bodies = document.querySelectorAll('.tab-body');
  for (var i = 0; i < bodies.length; i++) { bodies[i].classList.remove('active'); }
  var tabs = document.querySelectorAll('.tab');
  for (var i = 0; i < tabs.length; i++) { tabs[i].classList.remove('active'); }
  document.getElementById('tab-' + id).classList.add('active');
  el.classList.add('active');
  if (id === 'signoff') buildSummary();
}

function gd(id, i) { return dec[id + '_' + i] || ''; }

function sd(id, i, v) {
  var k = id + '_' + i;
  dec[k] = (dec[k] === v) ? '' : v;
  sync();
  exp[id] = true;
  render();
}

function gStatus(g) {
  var allEmpty = true;
  var anyRemove = false;
  var allFilled = true;
  for (var i = 0; i < g.members.length; i++) {
    var d = gd(g.name, i);
    if (d !== '') allEmpty = false;
    if (d === 'remove') anyRemove = true;
    if (d === '') allFilled = false;
  }
  if (allEmpty) return 'pending';
  if (anyRemove) return 'action';
  if (allFilled) return 'done';
  return 'pending';
}

function sync() {
  var rem = 0, decided = 0, grpDone = 0, total = 0;
  for (var gi = 0; gi < GROUPS.length; gi++) {
    var g = GROUPS[gi];
    var allFilled = true;
    for (var i = 0; i < g.members.length; i++) {
      var d = gd(g.name, i);
      total++;
      if (d === 'remove') rem++;
      if (d !== '') decided++;
      if (d === '') allFilled = false;
    }
    if (g.members.length > 0 && allFilled) grpDone++;
  }
  document.getElementById('s-remove').textContent = rem;
  document.getElementById('sb-remove').textContent = rem;
  document.getElementById('sb-done').textContent = grpDone;
  document.getElementById('s-pct').textContent = Math.round(decided / Math.max(total, 1) * 100) + '%';
  var allDone = (grpDone === GROUPS.length);
  var os = document.getElementById('overall-status');
  os.textContent = allDone ? 'Complete - ready for sign-off' : 'In progress';
  os.style.color = allDone ? '#2e7d32' : '#e65100';
}

function render() {
  var q = document.getElementById('search').value.toLowerCase();
  var fr = document.getElementById('f-role').value;
  var fs = document.getElementById('f-status').value;
  var filtered = [];
  for (var i = 0; i < GROUPS.length; i++) {
    var g = GROUPS[i];
    if (q && g.name.toLowerCase().indexOf(q) === -1) continue;
    if (fr && g.role !== fr) continue;
    if (fs && gStatus(g) !== fs) continue;
    filtered.push(g);
  }
  document.getElementById('cnt').textContent = 'Showing ' + filtered.length + ' of ' + GROUPS.length + ' groups';
  var html = '';
  for (var i = 0; i < filtered.length; i++) {
    var g = filtered[i];
    var open = exp[g.name] ? 'open' : '';
    var st = gStatus(g);
    var sbadge = st === 'done' ? '<span class="badge b-done">Reviewed</span>' :
                 st === 'action' ? '<span class="badge b-act">Action required</span>' :
                 '<span class="badge b-pend">Pending</span>';
    var done = 0, rem = 0;
    for (var mi = 0; mi < g.members.length; mi++) {
      var d = gd(g.name, mi);
      if (d !== '') done++;
      if (d === 'remove') rem++;
    }
    var pct = g.members.length ? Math.round(done / g.members.length * 100) : 0;
    var rows = '';
    for (var mi = 0; mi < g.members.length; mi++) {
      var m = g.members[mi];
      var d = gd(g.name, mi);
      var rc = d === 'keep' ? 'rk' : d === 'remove' ? 'rr' : '';
      var gn = g.name.replace(/'/g, "\\'");
      rows += '<tr class="' + rc + '"><td>' + m.upn + '</td><td>' + m.name + '</td><td>' + m.type + '</td><td>' + m.added + '</td>' +
        '<td><div class="dec-wrap">' +
        '<button class="dbtn ' + (d === 'keep' ? 'dk' : '') + '" onclick="sd(\'' + gn + '\',' + mi + ',\'keep\')">Keep</button>' +
        '<button class="dbtn ' + (d === 'remove' ? 'dr' : '') + '" onclick="sd(\'' + gn + '\',' + mi + ',\'remove\')">Remove</button>' +
        '</div></td></tr>';
    }
    var scopeTxt = g.scope ? ' <span style="color:#aaa;font-size:10px">' + g.scope + '</span>' : '';
    var roleHtml = g.role ? '<span class="badge b-role">' + g.role + '</span>' : '';
    html += '<div class="group">' +
      '<div class="ghdr" onclick="tog(\'' + g.name.replace(/'/g, "\\'") + '\')">' +
      '<span class="chev ' + open + '">&#9654;</span>' +
      '<span class="gname" title="' + g.name + '">' + g.name + '</span>' +
      '<span class="badge b-sub">' + g.sub + '</span>' +
      roleHtml +
      '<span class="badge b-count">' + g.members.length + ' members</span>' + sbadge + '</div>' +
      '<div class="desc">' + g.desc + scopeTxt + '</div>' +
      '<div class="panel ' + open + '">' +
      '<table><thead><tr><th>UPN</th><th>Display name</th><th>Type</th><th>Date added</th><th>Decision</th></tr></thead>' +
      '<tbody>' + rows + '</tbody></table>' +
      '<div class="footer-row"><span>' + done + '/' + g.members.length + ' decided' +
      (rem ? ' &nbsp;|&nbsp; <strong style="color:#b71c1c">' + rem + ' flagged for removal</strong>' : '') + '</span></div>' +
      '<div class="prog"><div class="prog-fill" style="width:' + pct + '%"></div></div>' +
      '</div></div>';
  }
  document.getElementById('list').innerHTML = html;

  // Populate role filter once
  var sel = document.getElementById('f-role');
  if (sel.options.length === 1) {
    var roles = [];
    for (var i = 0; i < GROUPS.length; i++) {
      var r = GROUPS[i].role;
      if (r && roles.indexOf(r) === -1) roles.push(r);
    }
    roles.sort();
    for (var i = 0; i < roles.length; i++) {
      var o = document.createElement('option');
      o.value = roles[i];
      o.textContent = roles[i];
      sel.appendChild(o);
    }
  }
}

function tog(id) { exp[id] = !exp[id]; render(); }
function expandAll() { for (var i = 0; i < GROUPS.length; i++) exp[GROUPS[i].name] = true; render(); }
function collapseAll() { exp = {}; render(); }

function buildSummary() {
  var rows = '';
  for (var gi = 0; gi < GROUPS.length; gi++) {
    var g = GROUPS[gi];
    for (var mi = 0; mi < g.members.length; mi++) {
      var m = g.members[mi];
      var d = gd(g.name, mi) || 'pending';
      var bg = d === 'keep' ? '#f1f8f1' : d === 'remove' ? '#fff5f5' : '';
      var fc = d === 'remove' ? '#b71c1c' : d === 'keep' ? '#2e7d32' : '#888';
      rows += '<tr style="background:' + bg + '"><td>' + g.name + '</td><td>' + m.upn + '</td><td>' + m.type + '</td>' +
        '<td><strong style="color:' + fc + '">' + d + '</strong></td><td>' + g.role + '</td></tr>';
    }
  }
  document.getElementById('summary-body').innerHTML = rows;
}

function exportCSV() {
  var rows = ['Group,Subscription,Azure Role,Scope,UPN,Display Name,Type,Decision,Review Date,Reviewer,Notes'];
  var notes = document.getElementById('so-notes').value.replace(/"/g, '""');
  for (var gi = 0; gi < GROUPS.length; gi++) {
    var g = GROUPS[gi];
    for (var mi = 0; mi < g.members.length; mi++) {
      var m = g.members[mi];
      var d = gd(g.name, mi) || 'pending';
      var fields = [g.name, g.sub, g.role, g.scope, m.upn, m.name, m.type, d, META.date, META.reviewer, notes];
      var line = '';
      for (var fi = 0; fi < fields.length; fi++) {
        if (fi > 0) line += ',';
        line += '"' + String(fields[fi] || '').replace(/"/g, '""') + '"';
      }
      rows.push(line);
    }
  }
  var a = document.createElement('a');
  a.href = 'data:text/csv;charset=utf-8,' + encodeURIComponent(rows.join('\n'));
  a.download = 'RBAC_Review_' + META.date + '.csv';
  a.click();
}
