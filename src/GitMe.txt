stages:
  - precheck
  - scan
  - test
  - ioc
  - report

variables:
  DEPENDENCY_SCANNING_STAGE: scan
  NPM_CONFIG_IGNORE_SCRIPTS: "true"
  ASTENABLEMRPIPELINES: "true"
  SECRETDETECTIONHISTORICSCAN: "true"

include:
  - template: Security/Dependency-Scanning.gitlab-ci.yml
  - template: Security/Secret-Detection.gitlab-ci.yml

precheck:
  stage: precheck
  image: alpine:3.19
  script:
    - |
      if [ ! -f package.json ]; then
        echo "No Node project detected"
        exit 0
      fi
      echo "Node project detected"
  rules:
    - exists:
        - package.json

ioc_shaihulud:
  stage: ioc
  image: alpine:3.19
  before_script:
    - apk add --no-cache ripgrep
  script:
    - mkdir -p ioc
    - echo "🔍 Scanning for Shai-Hulud IoCs"
    - rg -n 'setupbun\.js|bun_environment\.js|shai-hulud|trufflehog|preinstall|postinstall' . > ioc/shaihulud.txt || true
    - |
      if [ -s ioc/shaihulud.txt ]; then
        echo "🚨 IoC MATCHES FOUND"
        cat ioc/shaihulud.txt
        exit 1
      else
        echo "✅ No IoCs found"
      fi
  artifacts:
    when: always
    paths:
      - ioc/shaihulud.txt
  rules:
    - exists:
        - package.json

report:
  stage: report
  script:
    - echo "Scan completed"
  rules:
    - when: always
