# Requires Az.Accounts and Az.Compute modules
# Login to Azure
Connect-AzAccount

# Optional: select specific subscription
# Select-AzSubscription -SubscriptionName "Your-Subscription-Name"

# Define output CSV
$OutputFile = "C:\Temp\WSUS_Status_Report.csv"
$Results = @()

# Get all Windows VMs
$VMs = Get-AzVM | Where-Object { $_.StorageProfile.OsDisk.OsType -eq "Windows" }

foreach ($vm in $VMs) {
    Write-Host "Checking WSUS role on VM: $($vm.Name)" -ForegroundColor Cyan
    try {
        $Command = 'Get-WindowsFeature -Name UpdateServices | Select-Object Name, Installed'
        $Result = Invoke-AzVMRunCommand -ResourceGroupName $vm.ResourceGroupName -VMName $vm.Name -CommandId 'RunPowerShellScript' -ScriptString $Command -ErrorAction Stop

        $Output = ($Result.Value[0].Message | ConvertFrom-String | Select-Object -First 1)
        $Installed = if ($Output.P3 -eq 'True') { 'Installed' } else { 'Not Installed' }

        $Results += [PSCustomObject]@{
            VMName          = $vm.Name
            ResourceGroup   = $vm.ResourceGroupName
            Location        = $vm.Location
            WSUS_Status     = $Installed
        }
    }
    catch {
        Write-Warning "Failed to query $($vm.Name): $_"
        $Results += [PSCustomObject]@{
            VMName          = $vm.Name
            ResourceGroup   = $vm.ResourceGroupName
            Location        = $vm.Location
            WSUS_Status     = 'Error/No Access'
        }
    }
}

# Export report
$Results | Export-Csv -Path $OutputFile -NoTypeInformation
Write-Host "`nReport saved to: $OutputFile" -ForegroundColor Green

stages:

variables:
  ACR_REGISTRY: racholabacr.azurecr.io
  IMAGE_NAME: ihp-base-python
  IMAGE_PATH: base-images/ihp-base-python
  DOCKER_DRIVER: overlay2
  DOCKER_TLS_CERTDIR: ""
  DOCKER_HOST: tcp://docker:2375

build_python_image:
  stage: build
  image: docker:26.1.4
  services:
    - docker:dind
  script:
    - set -euo pipefail
    - VERSION=$(cat "$IMAGE_PATH/VERSION")
    - echo -n "$VERSION" > image_version.txt
    - docker build --pull --tag "$ACR_REGISTRY/$IMAGE_NAME:$VERSION" --tag "$ACR_REGISTRY/$IMAGE_NAME:latest" "$IMAGE_PATH"
    - docker save "$ACR_REGISTRY/$IMAGE_NAME:$VERSION" -o image.tar
  artifacts:
    paths:
      - image.tar
      - image_version.txt
    expire_in: 1 day

scan_python_image:
  stage: scan
  image: docker:26.1.4
  services:
    - docker:dind
  needs:
    - job: build_python_image
      artifacts: true
  before_script:
    - apk add --no-cache curl
    - curl -sfL https://raw.githubusercontent.com/aquasecurity/trivy/main/contrib/install.sh | sh -s -- -b /usr/local/bin
  script:
    - set -euo pipefail
    - VERSION=$(cat image_version.txt)
    - docker load -i image.tar
    - trivy image --exit-code 1 --severity HIGH,CRITICAL --scanners vuln "$ACR_REGISTRY/$IMAGE_NAME:$VERSION"
    - trivy image --format json --output trivy-report.json "$ACR_REGISTRY/$IMAGE_NAME:$VERSION"
    - trivy image --format spdx-json --output sbom.spdx.json "$ACR_REGISTRY/$IMAGE_NAME:$VERSION"
  artifacts:
    paths:
      - image.tar
      - image_version.txt
      - sbom.spdx.json
      - trivy-report.json
    expire_in: 7 days

publish_python_image:
  stage: publish
  image: docker:26.1.4
  services:
    - docker:dind
  needs:
    - job: scan_python_image
      artifacts: true
  script:
    - set -euo pipefail
    - VERSION=$(cat image_version.txt)
    - docker load -i image.tar
    - echo "$ACR_PASSWORD" | docker login "$ACR_REGISTRY" -u "$ACR_USERNAME" --password-stdin
    - docker tag "$ACR_REGISTRY/$IMAGE_NAME:$VERSION" "$ACR_REGISTRY/$IMAGE_NAME:latest"
    - docker push "$ACR_REGISTRY/$IMAGE_NAME:$VERSION"
    - docker push "$ACR_REGISTRY/$IMAGE_NAME:latest"
    - docker tag "$ACR_REGISTRY/$IMAGE_NAME:$VERSION" "$ACR_REGISTRY/$IMAGE_NAME:stable"
    - docker push "$ACR_REGISTRY/$IMAGE_NAME:stable"
