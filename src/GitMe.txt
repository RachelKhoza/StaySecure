aram(
    [Parameter(Mandatory = $true)]
    [string]$ResourceGroupName,
 
    [Parameter(Mandatory = $true)]
    [string]$VmName,
 
    [string]$AdoOrganization      = "anywhere365",
    [int]$IdleTimeoutMinutes      = 60,
    [int]$IdlePollSeconds         = 60,
    [int]$DrainWaitSeconds        = 25,
    [int]$VmBootTimeoutMinutes    = 30,
    [int]$VmPollSeconds           = 30,
    [int]$Stage2TimeoutMinutes    = 20,
 
    [switch]$CheckOnly
)
 
$ErrorActionPreference = "Stop"

$VmConfigMap = @{
    "EMEAWEUT00AZDE" = @{
        PoolId = 14
        Agents = @(
            "EMEAWEUT0001",
            "EMEAWEUT0002",
            "EMEAWEUT0003",

# =========================================================
# Helpers
# =========================================================
function Write-Log {
    param([string]$Message)
    Write-Host "[$(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')] $Message"
}
 
function New-AdoAuthHeader {
    param([string]$Pat)
    $base64AuthInfo = [Convert]::ToBase64String([Text.Encoding]::ASCII.GetBytes(":$Pat"))
    @{
        Authorization  = "Basic $base64AuthInfo"
        "Content-Type" = "application/json"
    }
}
 
function Get-VmPatchConfig {
    param(
        [string]$VmName,
        [hashtable]$VmConfigMap
    )
    if (-not $VmConfigMap.ContainsKey($VmName)) {
        throw "No patch config found for VM '$VmName'"
    }
    $VmConfigMap[$VmName]
}
 
function Get-AdoAgentsInPool {
    param(
        [string]$Organization,
        [int]$PoolId,
        [hashtable]$Headers
    )
    $uri = "https://dev.azure.com/$Organization/_apis/distributedtask/pools/$PoolId/agents?includeAssignedRequest=true&api-version=7.0"
    (Invoke-RestMethod -Uri $uri -Headers $Headers -Method Get).value
}
 
function Get-AdoAgentsForVm {
    param(
        [array]$Agents,
        [string[]]$ExpectedAgentNames
    )
    $matchedAgents = $Agents | Where-Object { $ExpectedAgentNames -contains $_.name }
    if (-not $matchedAgents -or $matchedAgents.Count -eq 0) {
        throw "No mapped ADO agents found in pool."
    }
    $matchedAgents
}
 
function Test-AdoAgentBusy {
    param($Agent)
    if ($null -ne $Agent.assignedRequest) { return $true }
    return $false
}
 
function Test-AllVmAgentsIdle {
    param([array]$VmAgents)
    foreach ($agent in $VmAgents) {
        if ($agent.status -ne "online") {
            Write-Log "Agent $($agent.name) is not online. Status: $($agent.status)"
            return $false
        }
        if (Test-AdoAgentBusy -Agent $agent) {
            Write-Log "Agent $($agent.name) is busy."
            return $false
        }
    }
    return $true
}
 
function Set-AdoAgentEnabledState {
    param(
        [string]$Organization,
        [int]$PoolId,
        [int]$AgentId,
        [string]$AgentName,
        [bool]$Enabled,
        [hashtable]$Headers
    )
    # Backtick fix: escape '?' so PowerShell doesn't mis-parse it
    $uri = "https://dev.azure.com/$Organization/_apis/distributedtask/pools/$PoolId/agents/$AgentId`?api-version=7.0"
    $body = @{
        id      = $AgentId
        enabled = $Enabled
        name    = $AgentName
    } | ConvertTo-Json
 
    Invoke-RestMethod -Uri $uri -Headers $Headers -Method Patch -Body $body | Out-Null
}
 
function Set-AllAgentsEnabledState {
    param(
        [array]$VmAgents,
        [string]$Organization,
        [int]$PoolId,
        [bool]$Enabled,
        [hashtable]$Headers
    )
    foreach ($agent in $VmAgents) {
        Write-Log "Setting agent $($agent.name) enabled=$Enabled"
        Set-AdoAgentEnabledState `
            -Organization $Organization `
            -PoolId $PoolId `
            -AgentId $agent.id `
            -AgentName $agent.name `
            -Enabled $Enabled `
            -Headers $Headers
    }
}
 
function Wait-ForVmGuestReady {
    param(
        [string]$ResourceGroupName,
        [string]$VmName,
        [int]$TimeoutMinutes = 30,
        [int]$PollSeconds    = 30
    )
    $deadline = (Get-Date).AddMinutes($TimeoutMinutes)
    while ((Get-Date) -lt $deadline) {
        try {
            $current = Get-AzVM -ResourceGroupName $ResourceGroupName -Name $VmName -Status
            $powerState = ($current.Statuses | Where-Object Code -like 'PowerState/*').DisplayStatus
 
            if ($powerState -eq "VM running") {
                $probe = Invoke-AzVMRunCommand `
                    -ResourceGroupName $ResourceGroupName `
                    -VMName $VmName `
                    -CommandId "RunPowerShellScript" `
                    -ScriptString 'Write-Output "VM_GUEST_READY"'
 
                $msg = ($probe.Value | ForEach-Object { $_.Message }) -join "`n"
                if ($msg -match "VM_GUEST_READY") {
                    return $true
                }
            }
        }
        catch {
            # ignore and keep polling
        }
        Start-Sleep -Seconds $PollSeconds
    }
    return $false
}
 
function Invoke-VmScriptWithRetry {
    <#
        Runs a script on the VM via Invoke-AzVMRunCommand with retry.
        Returns the joined stdout from VM messages.
    #>
    param(
        [string]$ResourceGroupName,
        [string]$VmName,
        [string]$ScriptString,
        [int]$MaxAttempts = 3,
        [int]$RetryDelaySeconds = 30
    )
    $attempt = 0
    $lastErr = $null
    while ($attempt -lt $MaxAttempts) {
        $attempt++
        try {
            $r = Invoke-AzVMRunCommand `
                -ResourceGroupName $ResourceGroupName `
                -VMName $VmName `
                -CommandId "RunPowerShellScript" `
                -ScriptString $ScriptString
            return ($r.Value | ForEach-Object { $_.Message }) -join "`n"
        }
        catch {
            $lastErr = $_
            Write-Log "Invoke-AzVMRunCommand attempt $attempt failed: $($_.Exception.Message)"
            if ($attempt -lt $MaxAttempts) { Start-Sleep -Seconds $RetryDelaySeconds }
        }
    }
    throw "Invoke-VmScriptWithRetry: all $MaxAttempts attempts failed. Last error: $($lastErr.Exception.Message)"
}
 
function Get-VmResultFile {
    <#
        Reads a result/log file from the VM. Returns string content (may be empty).
    #>
    param(
        [string]$ResourceGroupName,
        [string]$VmName,
        [string]$RemotePath
    )
    $script = @"
if (Test-Path '$RemotePath') {
    Get-Content '$RemotePath' -Raw
} else {
    Write-Output 'FILE_NOT_FOUND'
}
"@
    try {
        Invoke-VmScriptWithRetry `
            -ResourceGroupName $ResourceGroupName `
            -VmName $VmName `
            -ScriptString $script `
            -MaxAttempts 2 `
            -RetryDelaySeconds 15
    }
    catch {
        Write-Log "Get-VmResultFile failed: $($_.Exception.Message)"
        return ""
    }
}
 
# =========================================================
# Run-status flags (for summary)
# =========================================================
$AdoDrainAttempted     = $false
$AdoDrainSucceeded     = $false
$AdoReenableAttempted  = $false
$AdoReenableSucceeded  = $false
 
$Stage1Started   = $false
$Stage1Succeeded = $false
$VmRecovered     = $false
$Stage2Started   = $false
$Stage2Succeeded = $false
 
$OverallSuccess = $false
$ScriptError    = $null
 
# Captured for finally-block re-enable
$PoolId          = $null
$ExpectedAgents  = $null
$Headers         = $null
$vmAgents        = $null
 
try {
    # =========================================================
    # PAT / Config
    # =========================================================
    $AdoPat = $env:ADO_PAT
    if (-not $AdoPat) {
        throw "Environment variable ADO_PAT is not set."
    }
 
    $vmConfig        = Get-VmPatchConfig -VmName $VmName -VmConfigMap $VmConfigMap
    $PoolId          = $vmConfig.PoolId
    $ExpectedAgents  = $vmConfig.Agents
    $Headers         = New-AdoAuthHeader -Pat $AdoPat
 
    Write-Log "Resolved VM config: VM=$VmName PoolId=$PoolId Agents=$($ExpectedAgents -join ', ')"
 
    # =========================================================
    # VM Power Check
    # =========================================================
    $vmStatus   = Get-AzVM -ResourceGroupName $ResourceGroupName -Name $VmName -Status
    $powerState = ($vmStatus.Statuses | Where-Object Code -like "PowerState/*").DisplayStatus
 
    if ($powerState -ne "VM running") {
        throw "VM $VmName is not running. Current state: $powerState"
    }
    Write-Log "VM is running."
 
    # =========================================================
    # Wait for Idle
    # =========================================================
    Write-Log "Checking ADO agents for idle state..."
    $idleDeadline  = (Get-Date).AddMinutes($IdleTimeoutMinutes)
    $idleConfirmed = $false
    $vmAgents      = $null
 
    while ((Get-Date) -lt $idleDeadline) {
        $allAgents = Get-AdoAgentsInPool -Organization $AdoOrganization -PoolId $PoolId -Headers $Headers
        $vmAgents  = Get-AdoAgentsForVm -Agents $allAgents -ExpectedAgentNames $ExpectedAgents
 
        $agentSummary = $vmAgents | Select-Object name, status, enabled, @{Name="Busy";Expression={ $null -ne $_.assignedRequest }}
        $agentSummary | Format-Table -AutoSize | Out-String | Write-Host
 
        if (Test-AllVmAgentsIdle -VmAgents $vmAgents) {
            $idleConfirmed = $true
            break
        }
 
        Write-Log "Agents not idle yet. Waiting $IdlePollSeconds seconds..."
        Start-Sleep -Seconds $IdlePollSeconds
    }
 
    if (-not $idleConfirmed) {
        throw "Timed out waiting for all agents on VM $VmName to become idle."
    }
 
    Write-Log "All mapped agents are idle."
 
    if ($CheckOnly) {
        Write-Log "CheckOnly specified. Exiting before drain and patch."
        $OverallSuccess = $true
        return
    }
 
    # =========================================================
    # Drain ADO agents (BEST-EFFORT / SKIP-ON-ERROR)
    # =========================================================
    Write-Log "Disabling ADO agents to drain VM..."
    $AdoDrainAttempted = $true
    $AdoDrainSucceeded = $false
 
    try {
        Set-AllAgentsEnabledState `
            -VmAgents $vmAgents `
            -Organization $AdoOrganization `
            -PoolId $PoolId `
            -Enabled $false `
            -Headers $Headers
        $AdoDrainSucceeded = $true
    }
    catch {
        Write-Log "WARNING: Failed to disable ADO agents (continuing)."
        Write-Log "Details: $($_.Exception.Message)"
        $AdoDrainSucceeded = $false
    }
 
    if ($AdoDrainSucceeded) {
        Start-Sleep -Seconds $DrainWaitSeconds
 
        # Re-check after drain (BEST-EFFORT validation; do not block patching)
        Write-Log "Re-checking agents after drain..."
        try {
            $allAgents = Get-AdoAgentsInPool -Organization $AdoOrganization -PoolId $PoolId -Headers $Headers
            $vmAgents  = Get-AdoAgentsForVm -Agents $allAgents -ExpectedAgentNames $ExpectedAgents
 
            $agentSummary = $vmAgents | Select-Object name, status, enabled, @{Name="Busy";Expression={ $null -ne $_.assignedRequest }}
            $agentSummary | Format-Table -AutoSize | Out-String | Write-Host
 
            if (-not (Test-AllVmAgentsIdle -VmAgents $vmAgents)) {
                Write-Log "WARNING: Agents not idle after drain re-check (continuing)."
            } else {
                Write-Log "Drain successful."
            }
        }
        catch {
            Write-Log "WARNING: Drain re-check failed: $($_.Exception.Message)"
        }
    }
    else {
        Write-Log "Proceeding without confirmed ADO drain."
    }
 
    Write-Log "Proceeding to stop Azure Pipelines services and patch."
 
    # =========================================================
    # Stage 1: Stop services + baseline + patch
    # =========================================================
    $stage1Script = @'
param(
    [string]$LogFolder = "C:\PatchAutomation"
)
 
$ErrorActionPreference = "Stop"
New-Item -ItemType Directory -Force -Path $LogFolder | Out-Null
$logFile = Join-Path $LogFolder ("stage1-" + (Get-Date -Format "yyyyMMdd-HHmmss") + ".log")
 
function Write-Log {
    param([string]$Message)
    $line = "$(Get-Date -Format s) $Message"
    $line | Tee-Object -FilePath $logFile -Append | Out-Null
}
 
Write-Log "Capturing pre-patch PowerShell and module baseline"
 
$PSVersionTable | Out-File (Join-Path $LogFolder "psversion-before.txt") -Force
 
Get-Module -ListAvailable |
    Select-Object Name, Version |
    Sort-Object Name, Version -Unique |
    Export-Csv (Join-Path $LogFolder "modules-before.csv") -NoTypeInformation
 
Write-Log "Stopping Azure Pipelines services"
$svc = Get-Service | Where-Object { $_.DisplayName -like "Azure Pipelines*" }
 
if (-not $svc) {
    throw "No Azure Pipelines services found."
}
 
$svc | Select-Object Name, DisplayName, Status | Format-Table -AutoSize | Out-String |
    Out-File (Join-Path $LogFolder "services-before-stop.txt") -Force
 
foreach ($service in $svc) {
    if ($service.Status -ne "Stopped") {
        Stop-Service -Name $service.Name -Force
    }
}
 
$serviceStopDeadline = (Get-Date).AddMinutes(2)
$allStopped = $false
 
while ((Get-Date) -lt $serviceStopDeadline) {
    $currentServices = Get-Service | Where-Object { $_.DisplayName -like "Azure Pipelines*" }
    if (($currentServices | Where-Object { $_.Status -ne "Stopped" }).Count -eq 0) {
        $allStopped = $true
        break
    }
    Start-Sleep -Seconds 5
}
 
$currentServices = Get-Service | Where-Object { $_.DisplayName -like "Azure Pipelines*" }
$currentServices | Select-Object Name, DisplayName, Status | Format-Table -AutoSize | Out-String |
    Out-File (Join-Path $LogFolder "services-after-stop.txt") -Force
 
if (-not $allStopped) {
    throw "Not all Azure Pipelines services stopped within timeout."
}
 
Write-Log "All Azure Pipelines services are fully stopped."
 
Write-Log "Ensuring PSWindowsUpdate exists"
if (-not (Get-Module -ListAvailable -Name PSWindowsUpdate)) {
    Install-PackageProvider -Name NuGet -Force
    Install-Module -Name PSWindowsUpdate -Force -AllowClobber
}
 
Import-Module PSWindowsUpdate
 
Write-Log "Scanning for applicable Security updates"
$updates = Get-WindowsUpdate -Category "Security Updates" -MicrosoftUpdate
 
if (-not $updates) {
    Write-Log "No applicable security updates found"
    "NO_UPDATES" | Out-File (Join-Path $LogFolder "stage1-result.txt") -Force
    exit 0
}
 
$updates |
    Select-Object Title, KB, Size |
    Format-Table -AutoSize |
    Out-String |
    Out-File (Join-Path $LogFolder "available-security-updates.txt") -Force
 
Write-Log "Installing latest applicable Security updates"
Install-WindowsUpdate -Category "Security Updates" -MicrosoftUpdate -AcceptAll -IgnoreReboot |
    Tee-Object -FilePath (Join-Path $LogFolder "install-output.txt") -Append | Out-Null
 
$rebootRequired = Test-Path "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\WindowsUpdate\Auto Update\RebootRequired"
 
if ($rebootRequired) {
    Write-Log "Reboot required"
    "REBOOT_REQUIRED" | Out-File (Join-Path $LogFolder "stage1-result.txt") -Force
    Restart-Computer -Force
}
else {
    Write-Log "No reboot required"
    "COMPLETED_NO_REBOOT" | Out-File (Join-Path $LogFolder "stage1-result.txt") -Force
}
'@
 
    try {
        $Stage1Started = $true
        Write-Log "Starting Stage 1 on VM..."
        Invoke-AzVMRunCommand `
            -ResourceGroupName $ResourceGroupName `
            -VMName $VmName `
            -CommandId "RunPowerShellScript" `
            -ScriptString $stage1Script | Out-Null
 
        $Stage1Succeeded = $true
    }
    catch {
        Write-Log "Stage 1 failed."
        throw
    }
 
    # =========================================================
    # Wait for reboot/recovery
    # =========================================================
    Write-Log "Waiting for VM recovery..."
    $vmReady = Wait-ForVmGuestReady `
        -ResourceGroupName $ResourceGroupName `
        -VmName $VmName `
        -TimeoutMinutes $VmBootTimeoutMinutes `
        -PollSeconds $VmPollSeconds
 
    if (-not $vmReady) {
        throw "VM did not recover within timeout."
    }
 
    $VmRecovered = $true
    Write-Log "VM is back and guest commands are working."
 
    # =========================================================
    # Stage 2: baseline compare + start services
    # =========================================================
    $stage2Script = @'
param(
    [string]$LogFolder = "C:\PatchAutomation"
)
 
$ErrorActionPreference = "Stop"
$logFile = Join-Path $LogFolder ("stage2-" + (Get-Date -Format "yyyyMMdd-HHmmss") + ".log")
 
function Write-Log {
    param([string]$Message)
    $line = "$(Get-Date -Format s) $Message"
    $line | Tee-Object -FilePath $logFile -Append | Out-Null
}
 
# Always (re)write a starting marker so the orchestrator can tell Stage 2 began
"STAGE2_STARTED" | Out-File (Join-Path $LogFolder "stage2-result.txt") -Force
 
Write-Log "Capturing post-patch PowerShell and module state"
 
$PSVersionTable | Out-File (Join-Path $LogFolder "psversion-after.txt") -Force
 
Get-Module -ListAvailable |
    Select-Object Name, Version |
    Sort-Object Name, Version -Unique |
    Export-Csv (Join-Path $LogFolder "modules-after.csv") -NoTypeInformation
 
$beforePath = Join-Path $LogFolder "modules-before.csv"
$afterPath  = Join-Path $LogFolder "modules-after.csv"
 
if ((Test-Path $beforePath) -and (Test-Path $afterPath)) {
    $before = Import-Csv $beforePath
    $after  = Import-Csv $afterPath
 
    $diff = Compare-Object $before $after -Property Name, Version
    if ($diff) {
        $diff | Out-File (Join-Path $LogFolder "modules-diff.txt") -Force
        Write-Log "Module differences detected"
    } else {
        Write-Log "No module differences detected"
    }
}
 
# Reboot-still-required check BEFORE service start, so we exit clean
$rebootRequired = Test-Path "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\WindowsUpdate\Auto Update\RebootRequired"
if ($rebootRequired) {
    "REBOOT_STILL_REQUIRED" | Out-File (Join-Path $LogFolder "stage2-result.txt") -Force
    Write-Log "Reboot still required after patch"
    Write-Output "REBOOT_STILL_REQUIRED"
    exit 3
}
 
Write-Log "Starting Azure Pipelines services"
$svc = Get-Service | Where-Object { $_.DisplayName -like "Azure Pipelines*" }
 
if (-not $svc) {
    "NO_SERVICES_FOUND" | Out-File (Join-Path $LogFolder "stage2-result.txt") -Force
    throw "No Azure Pipelines services found."
}
 
foreach ($service in $svc) {
    if ($service.Status -ne "Running") {
        Start-Service -Name $service.Name
    }
}
 
$serviceStartDeadline = (Get-Date).AddMinutes(2)
$allRunning = $false
 
while ((Get-Date) -lt $serviceStartDeadline) {
    $currentServices = Get-Service | Where-Object { $_.DisplayName -like "Azure Pipelines*" }
    if (($currentServices | Where-Object { $_.Status -ne "Running" }).Count -eq 0) {
        $allRunning = $true
        break
    }
    Start-Sleep -Seconds 5
}
 
$currentServices = Get-Service | Where-Object { $_.DisplayName -like "Azure Pipelines*" }
$currentServices | Select-Object Name, DisplayName, Status | Format-Table -AutoSize | Out-String |
    Out-File (Join-Path $LogFolder "services-after-start.txt") -Force
 
if (-not $allRunning) {
    "SERVICES_FAILED_TO_START" | Out-File (Join-Path $LogFolder "stage2-result.txt") -Force
    throw "Not all Azure Pipelines services started within timeout."
}
 
Write-Log "All Azure Pipelines services are fully running."
 
# IMPORTANT: write success marker to disk BEFORE emitting to stdout.
# Invoke-AzVMRunCommand can truncate stdout; the result file is the source of truth.
"SERVICE_VALIDATION_COMPLETE" | Out-File (Join-Path $LogFolder "stage2-result.txt") -Force
Write-Output "SERVICE_VALIDATION_COMPLETE"
'@
 
    try {
        $Stage2Started = $true
        Write-Log "Starting Stage 2 on VM..."
 
        $stage2Output = Invoke-VmScriptWithRetry `
            -ResourceGroupName $ResourceGroupName `
            -VmName $VmName `
            -ScriptString $stage2Script `
            -MaxAttempts 2 `
            -RetryDelaySeconds 30
 
        # Primary check: stdout
        if ($stage2Output -match "SERVICE_VALIDATION_COMPLETE") {
            $Stage2Succeeded = $true
        }
        elseif ($stage2Output -match "REBOOT_STILL_REQUIRED") {
            throw "Stage 2: reboot still required after patching."
        }
        else {
            # Fallback check: read result file from the VM (handles Invoke-AzVMRunCommand stdout truncation)
            Write-Log "Stage 2 stdout did not contain success marker. Checking result file on VM..."
            Start-Sleep -Seconds 10
            $resultContent = Get-VmResultFile `
                -ResourceGroupName $ResourceGroupName `
                -VmName $VmName `
                -RemotePath "C:\PatchAutomation\stage2-result.txt"
 
            Write-Log "stage2-result.txt content: $resultContent"
 
            if ($resultContent -match "SERVICE_VALIDATION_COMPLETE") {
                Write-Log "Stage 2 succeeded (confirmed via result file on VM)."
                $Stage2Succeeded = $true
            }
            elseif ($resultContent -match "REBOOT_STILL_REQUIRED") {
                throw "Stage 2: reboot still required after patching (confirmed via result file)."
            }
            elseif ($resultContent -match "SERVICES_FAILED_TO_START") {
                throw "Stage 2: services failed to start (confirmed via result file)."
            }
            else {
                # Last-chance poll: services may have started but marker not yet written
                $deadline = (Get-Date).AddMinutes($Stage2TimeoutMinutes)
                while ((Get-Date) -lt $deadline) {
                    Start-Sleep -Seconds 30
                    $resultContent = Get-VmResultFile `
                        -ResourceGroupName $ResourceGroupName `
                        -VmName $VmName `
                        -RemotePath "C:\PatchAutomation\stage2-result.txt"
                    if ($resultContent -match "SERVICE_VALIDATION_COMPLETE") {
                        Write-Log "Stage 2 succeeded (confirmed via delayed result file poll)."
                        $Stage2Succeeded = $true
                        break
                    }
                }
 
                if (-not $Stage2Succeeded) {
                    throw "Stage 2 validation did not confirm success. Last result: $resultContent"
                }
            }
        }
    }
    catch {
        Write-Log "Stage 2 failed: $($_.Exception.Message)"
        throw
    }
 
    Write-Log "Patch flow completed successfully."
    Write-Host ""
    Write-Host "Artifacts on VM should be under: C:\PatchAutomation"
 
    $OverallSuccess = $true
}
catch {
    $ScriptError = $_
    Write-Log "ERROR: Script failed: $($_.Exception.Message)"
    # Do NOT rethrow yet - finally must run to re-enable agents
}
finally {
    # =========================================================
    # Re-enable ADO agents (BEST-EFFORT, ALWAYS-RUN)
    # This block runs even on failure so agents don't stay disabled.
    # =========================================================
    if ($AdoDrainAttempted -and $null -ne $vmAgents -and $null -ne $Headers -and $null -ne $PoolId) {
        Write-Log "Re-enabling ADO agents..."
        $AdoReenableAttempted = $true
        $AdoReenableSucceeded = $false
 
        # Refresh agent list (IDs and names should be stable, but be safe)
        try {
            $allAgents = Get-AdoAgentsInPool -Organization $AdoOrganization -PoolId $PoolId -Headers $Headers
            $vmAgents  = Get-AdoAgentsForVm -Agents $allAgents -ExpectedAgentNames $ExpectedAgents
        }
        catch {
            Write-Log "WARNING: Could not refresh agent list before re-enable: $($_.Exception.Message)"
        }
 
        # Try each agent individually so one failure doesn't skip the others
        $reenableFailures = @()
        foreach ($agent in $vmAgents) {
            $perAgentAttempts = 0
            $perAgentMax      = 3
            $perAgentOk       = $false
            while ($perAgentAttempts -lt $perAgentMax -and -not $perAgentOk) {
                $perAgentAttempts++
                try {
                    Write-Log "Re-enabling agent $($agent.name) (attempt $perAgentAttempts)"
                    Set-AdoAgentEnabledState `
                        -Organization $AdoOrganization `
                        -PoolId $PoolId `
                        -AgentId $agent.id `
                        -AgentName $agent.name `
                        -Enabled $true `
                        -Headers $Headers
                    $perAgentOk = $true
                }
                catch {
                    Write-Log "WARNING: Failed to re-enable $($agent.name): $($_.Exception.Message)"
                    if ($perAgentAttempts -lt $perAgentMax) { Start-Sleep -Seconds 10 }
                }
            }
            if (-not $perAgentOk) {
                $reenableFailures += $agent.name
            }
        }
 
        if ($reenableFailures.Count -eq 0) {
            $AdoReenableSucceeded = $true
            Write-Log "All agents re-enabled successfully."
        }
        else {
            Write-Log "WARNING: The following agents could not be re-enabled: $($reenableFailures -join ', ')"
            Write-Log "Re-enable them manually in Azure DevOps -> Agent Pool $PoolId."
            $AdoReenableSucceeded = $false
        }
    }
    elseif ($AdoDrainAttempted) {
        Write-Log "WARNING: Drain was attempted but agent context is incomplete; cannot auto re-enable."
    }
 
    # =========================================================
    # SUMMARY (always prints)
    # =========================================================
    Write-Host ""
    Write-Host "========== PATCH SUMMARY =========="
 
    $summary = [ordered]@{
        VmName                = $VmName
        ResourceGroup         = $ResourceGroupName
        AdoOrganization       = $AdoOrganization
        AgentPoolId           = $(if ($PoolId) { $PoolId } else { "" })
        Agents                = $(if ($ExpectedAgents) { ($ExpectedAgents -join ", ") } else { "" })
        CheckOnlyMode         = [bool]$CheckOnly
 
        AdoDrainAttempted     = [bool]$AdoDrainAttempted
        AdoDrainSucceeded     = [bool]$AdoDrainSucceeded
        AdoReenableAttempted  = [bool]$AdoReenableAttempted
        AdoReenableSucceeded  = [bool]$AdoReenableSucceeded
 
        Stage1Started         = [bool]$Stage1Started
        Stage1Succeeded       = [bool]$Stage1Succeeded
        VmRecovered           = [bool]$VmRecovered
        Stage2Started         = [bool]$Stage2Started
        Stage2Succeeded       = [bool]$Stage2Succeeded
 
        ArtifactsOnVm         = "C:\PatchAutomation"
        OverallSuccess        = [bool]$OverallSuccess
        Error                 = $(if ($ScriptError) { $ScriptError.Exception.Message } else { "" })
        CompletionTime        = (Get-Date).ToString("yyyy-MM-dd HH:mm:ss")
    }
 
    $summary.GetEnumerator() | ForEach-Object {
        "{0,-22}: {1}" -f $_.Key, $_.Value
    } | Write-Host
 
    Write-Host "==================================="
    Write-Host ""
 
    # If we caught an error earlier, surface it as a non-terminating warning
    # rather than rethrow, so the caller can rely on $OverallSuccess and the
    # exit code to drive automation. Flip to `throw $ScriptError` if you'd
    # rather propagate the exception.
    if ($ScriptError -and -not $OverallSuccess) {
        Write-Host ""
        Write-Host "Script ended with error: $($ScriptError.Exception.Message)" -ForegroundColor Red
        exit 1
    }
}
