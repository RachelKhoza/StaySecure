param(
    [Parameter(Mandatory = $true)]
    [string]$AdoOrganization,

    [Parameter(Mandatory = $true)]
    [int]$AdoPoolId,

    [Parameter(Mandatory = $true)]
    [string]$AdoPat,

    [Parameter(Mandatory = $true)]
    [string]$VmName
)

$ErrorActionPreference = 'Stop'

function New-AdoAuthHeader {
    param(
        [Parameter(Mandatory = $true)]
        [string]$Pat
    )

    $base64AuthInfo = [Convert]::ToBase64String(
        [Text.Encoding]::ASCII.GetBytes(":$Pat")
    )

    return @{
        Authorization = "Basic $base64AuthInfo"
        "Content-Type" = "application/json"
    }
}

function Get-AdoAgentsInPool {
    param(
        [Parameter(Mandatory = $true)]
        [string]$Organization,

        [Parameter(Mandatory = $true)]
        [int]$PoolId,

        [Parameter(Mandatory = $true)]
        [hashtable]$Headers
    )

    $uri = "https://dev.azure.com/$Organization/_apis/distributedtask/pools/$PoolId/agents?includeAssignedRequest=true&includeCapabilities=true&api-version=7.1"

    $response = Invoke-RestMethod -Uri $uri -Headers $Headers -Method Get
    return $response.value
}

function Get-AdoAgentsForVm {
    param(
        [Parameter(Mandatory = $true)]
        [array]$Agents,

        [Parameter(Mandatory = $true)]
        [string]$VmName
    )

    # Starter rule: agent name contains VM name
    return $Agents | Where-Object { $_.name -like "*$VmName*" }
}

function Test-AgentIdle {
    param(
        [Parameter(Mandatory = $true)]
        $Agent
    )

    if (-not $Agent.enabled) {
        return @{
            IsIdle = $false
            Reason = "Disabled"
        }
    }

    if ($Agent.status -ne "online") {
        return @{
            IsIdle = $false
            Reason = "Not online"
        }
    }

    if ($null -ne $Agent.assignedRequest) {
        return @{
            IsIdle = $false
            Reason = "Busy"
        }
    }

    return @{
        IsIdle = $true
        Reason = "Idle"
    }
}

try {
    $headers = New-AdoAuthHeader -Pat $AdoPat
    $allAgents = Get-AdoAgentsInPool -Organization $AdoOrganization -PoolId $AdoPoolId -Headers $headers
    $vmAgents = Get-AdoAgentsForVm -Agents $allAgents -VmName $VmName

    if (-not $vmAgents -or $vmAgents.Count -eq 0) {
        Write-Host "Status: No agents found for VM '$VmName'"
        exit 1
    }

    $allIdle = $true

    foreach ($agent in $vmAgents) {
        $result = Test-AgentIdle -Agent $agent

        Write-Host "Agent: $($agent.name)"
        Write-Host "  Enabled: $($agent.enabled)"
        Write-Host "  Status : $($agent.status)"
        Write-Host "  State  : $($result.Reason)"

        if (-not $result.IsIdle) {
            $allIdle = $false
        }
    }

    if ($allIdle) {
        Write-Host ""
        Write-Host "Status: Idle ready to patch"
        exit 0
    }
    else {
        Write-Host ""
        Write-Host "Status: Not ready to patch"
        exit 2
    }
}
catch {
    Write-Host "Status: Check failed"
    Write-Host "Error: $($_.Exception.Message)"
    exit 99
}
