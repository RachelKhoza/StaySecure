stages:
  - precheck
  - scan
  - ioc
  - report

# Global variables for containment
variables:
  NPM_CONFIG_IGNORE_SCRIPTS: "true"    # disable preinstall/postinstall execution
  ASTENABLEMRPIPELINES: "true"         # show findings in MRs
  SECRETDETECTIONHISTORICSCAN: "true"  # full history scan

# -------- 1. PRECHECK: Detect package.json without executing scripts -------
precheck:
  stage: precheck
  image: node:20-alpine
  script:
    - if [ ! -f package.json ]; then echo "No Node project"; exit 0; fi
  rules:
    - exists:
        - package.json

# -------- 2. SCAN: SBOM + Dependency Scan + Secrets Detection --------------
sbom_scan:
  stage: scan
  extends: .cyclonedx-sbom
  rules:
    - exists:
        - package-lock.json

dependency_scan:
  stage: scan
  extends: .dependency_scanning
  rules:
    - exists:
        - package-lock.json

secret_scan:
  stage: scan
  extends: .secret_detection
  rules:
    - when: always

# -------- 3. IoC Hunt for Shai-Hulud Malware -------------------------------
ioc_shaihulud:
  stage: ioc
  image: alpine:3.19
  before_script:
    - apk add --no-cache ripgrep git
  script:
    # Files malware drops
    - rg -n --glob 'node_modules/' 'setupbun.js|bun_environment.js' || true

    # Preinstall hooks (review carefully)
    - rg -n --glob 'node_modules/' '"preinstall"' || true

    # Rogue GitHub Actions files
    - rg -n --glob 'node_modules/' 'shai-hulud*' || true

    # References to TruffleHog secret harvesting
    - rg -n --glob 'node_modules/' 'trufflehog' || true

  artifacts:
    when: always
    paths: ["iocshaihulud.log"]
  allow_failure: false

# -------- 4. REPORT STAGE ---------------------------------------------------
report:
  stage: report
  script:
    - echo "IoC and SBOM Scan Completed"
  rules:
    - when: always
