param (
    [string]$CsvPath = ".\vms.csv",
    [string]$OutputPath = ".\patch-results.csv"
)

# Pre-req:
# Connect-AzAccount
# Set-AzContext -Subscription "your-sub-id"

$vms = Import-Csv -Path $CsvPath
$results = @()

foreach ($vmEntry in $vms) {

    $vmName = $vmEntry.VMName
    Write-Host "Processing VM: $vmName" -ForegroundColor Cyan

    # Get VM
    $vm = Get-AzVM | Where-Object { $_.Name -eq $vmName }

    if (-not $vm) {
        Write-Host "VM not found: $vmName" -ForegroundColor Red

        $results += [PSCustomObject]@{
            VMName              = $vmName
            Status              = "NotFound"
            MissingSecurityKbs  = ""
            SecurityUpdateCount = ""
            LastPatchedDate     = ""
            LastPatchedKB       = ""
            RebootRequired      = ""
        }

        continue
    }

    $rg = $vm.ResourceGroupName

    # Script executed INSIDE VM
    $script = @"
try {
    # Pending updates
    \$session = New-Object -ComObject Microsoft.Update.Session
    \$searcher = \$session.CreateUpdateSearcher()
    \$result = \$searcher.Search("IsInstalled=0 and Type='Software'")

    # Filter Security Updates
    \$securityUpdates = @()

    foreach (\$update in \$result.Updates) {
        foreach (\$cat in \$update.Categories) {
            if (\$cat.Name -eq "Security Updates") {
                \$kb = (\$update.KBArticleIDs -join ",")
                \$securityUpdates += [PSCustomObject]@{
                    KB    = \$kb
                    Title = \$update.Title
                }
                break
            }
        }
    }

    # Last installed update (fallback-safe)
    \$lastUpdate = Get-HotFix | Sort-Object InstalledOn -Descending | Select-Object -First 1

    # Reboot check
    \$rebootRequired = Test-Path "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\WindowsUpdate\Auto Update\RebootRequired"

    # Output
    [PSCustomObject]@{
        Status              = "Success"
        SecurityUpdateCount = \$securityUpdates.Count
        SecurityUpdates     = \$securityUpdates
        LastPatchedDate     = \$lastUpdate.InstalledOn
        LastPatchedKB       = \$lastUpdate.HotFixID
        RebootRequired      = \$rebootRequired
    } | ConvertTo-Json -Depth 5
}
catch {
    [PSCustomObject]@{
        Status = "Error"
        Error  = \$_ | Out-String
    } | ConvertTo-Json -Depth 3
}
"@

    try {
        $runCommand = Invoke-AzVMRunCommand `
            -ResourceGroupName $rg `
            -VMName $vmName `
            -CommandId 'RunPowerShellScript' `
            -ScriptString $script

        $raw = $runCommand.Value[0].Message

        if (-not $raw) {
            throw "Empty response"
        }

        $output = $raw | ConvertFrom-Json

        if ($output.Status -ne "Success") {
            throw "VM script failed"
        }

        $kbList = ($output.SecurityUpdates | ForEach-Object { $_.KB }) -join "; "

        $results += [PSCustomObject]@{
            VMName              = $vmName
            Status              = "Success"
            MissingSecurityKbs  = $kbList
            SecurityUpdateCount = $output.SecurityUpdateCount
            LastPatchedDate     = $output.LastPatchedDate
            LastPatchedKB       = $output.LastPatchedKB
            RebootRequired      = $output.RebootRequired
        }

    }
    catch {
        Write-Host "Error processing $vmName: $_" -ForegroundColor Yellow

        $results += [PSCustomObject]@{
            VMName              = $vmName
            Status              = "Error"
            MissingSecurityKbs  = ""
            SecurityUpdateCount = ""
            LastPatchedDate     = ""
            LastPatchedKB       = ""
            RebootRequired      = ""
        }
    }
}

# Display nicely
$results | Format-Table -AutoSize

# Export
$results | Export-Csv -Path $OutputPath -NoTypeInformation

Write-Host "`nResults exported to $OutputPath" -ForegroundColor Green
