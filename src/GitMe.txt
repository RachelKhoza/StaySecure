
param(
    [string]$AdoOrganization = "Any",
    [int]$PoolId = 117
)

$ErrorActionPreference = 'Stop'

# PAT must already exist in env var
$AdoPat = $env:ADO_PAT
if (-not $AdoPat) {
    throw "Environment variable ADO_PAT is not set."
}

# Agents you want to track
$TrackedAgents = @(
    "GLBLSEAP01",
    "GLBLSEAP02",
    "GLBLSEAP03",
    "GLBLSEAP04"
)

function New-AdoAuthHeader {
    param([Parameter(Mandatory = $true)][string]$Pat)

    $base64AuthInfo = [Convert]::ToBase64String([Text.Encoding]::ASCII.GetBytes(":$Pat"))
    return @{
        Authorization = "Basic $base64AuthInfo"
        "Content-Type" = "application/json"
    }
}

function Get-AdoAgentsInPool {
    param(
        [Parameter(Mandatory = $true)][string]$Organization,
        [Parameter(Mandatory = $true)][int]$PoolId,
        [Parameter(Mandatory = $true)][hashtable]$Headers
    )

    $uri = "https://dev.azure.com/$Organization/_apis/distributedtask/pools/$PoolId/agents?includeAssignedRequest=true&includeCapabilities=true&api-version=7.1"
    return (Invoke-RestMethod -Uri $uri -Headers $Headers -Method Get).value
}

function Get-AgentWorkState {
    param([Parameter(Mandatory = $true)]$Agent)

    if ($null -ne $Agent.assignedRequest) {
        return "Busy"
    }
    return "Idle"
}

$headers = New-AdoAuthHeader -Pat $AdoPat
$allAgents = Get-AdoAgentsInPool -Organization $AdoOrganization -PoolId $PoolId -Headers $headers

$matchedAgents = $allAgents | Where-Object { $TrackedAgents -contains $_.name } | Sort-Object name

if (-not $matchedAgents -or $matchedAgents.Count -eq 0) {
    throw "No tracked agents were found in pool $PoolId."
}

$report = foreach ($agent in $matchedAgents) {
    [PSCustomObject]@{
        AgentName   = $agent.name
        AgentId     = $agent.id
        Status      = $agent.status
        Enabled     = $agent.enabled
        WorkState   = Get-AgentWorkState -Agent $agent
        CurrentJob  = if ($null -ne $agent.assignedRequest) { $agent.assignedRequest.requestId } else { $null }
    }
}

$report | Format-Table -AutoSize
