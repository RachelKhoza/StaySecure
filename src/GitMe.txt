# ── CONFIG — edit these 3 lines ───────────────────────────────────────────────
$ReviewedBy     = "Cl"
$SubscriptionId = "xxxxxxxx-xxxx-xxxx-xxxx-xxxxxxxxxxxx"
$GroupPrefixes  = @("DR-", "NO-", "MG-")
# ─────────────────────────────────────────────────────────────────────────────
 
$ReviewDate = Get-Date -Format "yyyy-MM-dd"
$OutputPath = ".\RBAC_Review_$ReviewDate.html"
 
# ── CONNECT ───────────────────────────────────────────────────────────────────
Write-Host "`n[1/5] Connecting to Azure..." -ForegroundColor Cyan
Connect-AzAccount -SubscriptionId $SubscriptionId | Out-Null
Set-AzContext    -SubscriptionId $SubscriptionId  | Out-Null
$SubName = (Get-AzContext).Subscription.Name
Write-Host "      OK — $SubName" -ForegroundColor Green
 
Write-Host "[2/5] Connecting to Microsoft Graph..." -ForegroundColor Cyan
$Scopes = @(
    "Group.Read.All",
    "GroupMember.Read.All",
    "User.Read.All",
    "RoleManagement.Read.All",
    "AuditLog.Read.All"
)
Connect-MgGraph -Scopes $Scopes | Out-Null
Write-Host "      OK" -ForegroundColor Green
 
# ── PULL GROUPS + MEMBERS ─────────────────────────────────────────────────────
Write-Host "[3/5] Pulling security groups and members..." -ForegroundColor Cyan
$allGroups = Get-MgGroup -Filter "securityEnabled eq true" -All `
             -Property "DisplayName,Id,Description" |
             Where-Object {
                 $n = $_.DisplayName
                 $GroupPrefixes | Where-Object { $n -like "$_*" }
             }
Write-Host "      Found $($allGroups.Count) groups" -ForegroundColor Green
 
$roleAssignments = Get-AzRoleAssignment |
                   Where-Object { $_.ObjectType -eq "Group" } |
                   Select-Object DisplayName, RoleDefinitionName, Scope
 
$groupsData = [System.Collections.Generic.List[hashtable]]::new()
 
foreach ($g in $allGroups) {
    Write-Host "      $($g.DisplayName)" -ForegroundColor Gray
    $role  = $roleAssignments | Where-Object { $_.DisplayName -eq $g.DisplayName } | Select-Object -First 1
    $rawM  = Get-MgGroupMember -GroupId $g.Id -All -ErrorAction SilentlyContinue
    $members = [System.Collections.Generic.List[hashtable]]::new()
    foreach ($m in $rawM) {
        $user = Get-MgUser -UserId $m.Id -Property "UserPrincipalName,DisplayName" -ErrorAction SilentlyContinue
        if ($user) {
            $members.Add(@{ upn=$user.UserPrincipalName; name=$user.DisplayName; type="User"; added=($m.AdditionalProperties["createdDateTime"] ?? "Unknown") })
        } else {
            $sp = Get-MgServicePrincipal -ServicePrincipalId $m.Id -ErrorAction SilentlyContinue
            if ($sp) { $members.Add(@{ upn=$sp.AppId; name=$sp.DisplayName; type="Service Principal"; added="Unknown" }) }
        }
    }
    $groupsData.Add(@{
        name    = $g.DisplayName
        desc    = ($g.Description ?? "")
        role    = ($role ? $role.RoleDefinitionName : "No direct assignment")
        scope   = ($role ? $role.Scope : "")
        sub     = $SubName
        members = $members.ToArray()
    })
}
 
# ── PULL PIM DATA ─────────────────────────────────────────────────────────────
Write-Host "[4/5] Pulling PIM data..." -ForegroundColor Cyan
 
# Helper — resolve principal UPN from GUID
function Resolve-Principal($id) {
    $u = Get-MgUser -UserId $id -ErrorAction SilentlyContinue
    if ($u) { return $u.UserPrincipalName }
    $sp = Get-MgServicePrincipal -ServicePrincipalId $id -ErrorAction SilentlyContinue
    if ($sp) { return $sp.DisplayName }
    return $id
}
 
# Helper — resolve role name from definition ID
function Resolve-Role($id) {
    $r = Get-MgRoleManagementDirectoryRoleDefinition -UnifiedRoleDefinitionId $id -ErrorAction SilentlyContinue
    return ($r ? $r.DisplayName : $id)
}
 
# Active PIM assignments right now
$pimActiveData = [System.Collections.Generic.List[hashtable]]::new()
try {
    $active = Get-MgRoleManagementDirectoryRoleAssignmentScheduleInstance -All -ErrorAction Stop |
              Where-Object { $_.AssignmentType -eq "Activated" }
    foreach ($a in $active) {
        $pimActiveData.Add(@{
            principal  = Resolve-Principal $a.PrincipalId
            role       = Resolve-Role $a.RoleDefinitionId
            scope      = $a.DirectoryScopeId ?? "/"
            activatedAt= ($a.StartDateTime ? $a.StartDateTime.ToString("yyyy-MM-ddTHH:mm:ssZ") : "")
            expiresAt  = ($a.EndDateTime   ? $a.EndDateTime.ToString("yyyy-MM-ddTHH:mm:ssZ") : "")
            duration   = if ($a.StartDateTime -and $a.EndDateTime) {
                             "$([Math]::Round(($a.EndDateTime - $a.StartDateTime).TotalHours,1))h"
                         } else { "Unknown" }
        })
    }
} catch { Write-Host "      PIM active: $($_.Exception.Message)" -ForegroundColor Yellow }
 
# Eligible (JIT) assignments
$pimEligData = [System.Collections.Generic.List[hashtable]]::new()
try {
    $elig = Get-MgRoleManagementDirectoryRoleEligibilitySchedule -All -ErrorAction Stop
    foreach ($e in $elig) {
        $roleName = Resolve-Role $e.RoleDefinitionId
        $tier = switch -Regex ($roleName) {
            "Owner|Global Admin|Privileged"  { "Critical" }
            "Contributor|User Access Admin"  { "High" }
            "Reader|Monitor"                 { "Low" }
            default                          { "Medium" }
        }
        $pimEligData.Add(@{
            principal   = Resolve-Principal $e.PrincipalId
            role        = $roleName
            tier        = $tier
            maxDuration = ($e.ScheduleInfo.Expiration.Duration ?? "Not set")
            approval    = "Check PIM policy"
            mfa         = "Check PIM policy"
        })
    }
} catch { Write-Host "      PIM eligible: $($_.Exception.Message)" -ForegroundColor Yellow }
 
# PIM activation history — last 30 days via Audit Log
$pimHistData = [System.Collections.Generic.List[hashtable]]::new()
try {
    $since  = ((Get-Date).AddDays(-30)).ToString("yyyy-MM-ddTHH:mm:ssZ")
    $audits = Get-MgAuditLogDirectoryAudit -All -ErrorAction Stop |
              Where-Object { $_.LoggedByService -eq "PIM" -and $_.ActivityDateTime -ge $since }
    foreach ($a in $audits) {
        $principal = ($a.InitiatedBy.User.UserPrincipalName ?? $a.InitiatedBy.App.DisplayName ?? "Unknown")
        $role      = (($a.TargetResources | Where-Object { $_.Type -eq "Role" } | Select-Object -First 1).DisplayName ?? "Unknown")
        $just      = (($a.AdditionalDetails | Where-Object { $_.Key -eq "justification" }).Value ?? "")
        $approver  = (($a.AdditionalDetails | Where-Object { $_.Key -eq "approver" }).Value ?? "N/A")
        $pimHistData.Add(@{
            date          = $a.ActivityDateTime.ToString("yyyy-MM-dd HH:mm")
            principal     = $principal
            role          = $role
            justification = $just
            approvedBy    = $approver
            result        = $a.Result
        })
    }
} catch { Write-Host "      PIM history: $($_.Exception.Message)" -ForegroundColor Yellow }
 
Write-Host "      PIM active=$($pimActiveData.Count) eligible=$($pimEligData.Count) history=$($pimHistData.Count)" -ForegroundColor Green
 
# ── SERIALISE TO JSON ─────────────────────────────────────────────────────────
function To-Json($obj) { $obj | ConvertTo-Json -Depth 6 -Compress }
 
$groupsJson  = To-Json $groupsData.ToArray()
$pimActJson  = To-Json $pimActiveData.ToArray()
$pimEligJson = To-Json $pimEligData.ToArray()
$pimHistJson = To-Json $pimHistData.ToArray()
 
$pimActiveCnt  = $pimActiveData.Count
$pimEligCnt    = $pimEligData.Count
$pimHistCnt    = $pimHistData.Count
$pimFailCnt    = ($pimHistData | Where-Object { $_.result -eq "failure" }).Count
$pimCritCnt    = ($pimEligData | Where-Object { $_.tier -eq "Critical" }).Count
$totalMembers  = ($groupsData  | ForEach-Object { $_.members.Count } | Measure-Object -Sum).Sum
 
# ── HELPER: build table rows ──────────────────────────────────────────────────
function Badge($text, $cls) { "<span class='badge $cls'>$text</span>" }
function TierBadge($t) {
    $cls = @{Critical="b-crit";High="b-high";Medium="b-med";Low="b-low"}[$t] ?? "b-med"
    Badge $t $cls
}
function ResultBadge($r) {
    if ($r -eq "success") { Badge "Success" "b-done" } else { Badge "Failed" "b-act" }
}
 
$pimActiveRows = if ($pimActiveData.Count) {
    ($pimActiveData | ForEach-Object {
        "<tr><td>$($_.principal)</td><td>$($_.role)</td>" +
        "<td style='font-size:10px;color:#888'>$($_.scope)</td>" +
        "<td>$($_.activatedAt)</td><td>$($_.expiresAt)</td>" +
        "<td>$(Badge $_.duration 'b-act')</td></tr>"
    }) -join ""
} else { "<tr><td colspan='6' style='text-align:center;color:#888;padding:16px'>No active PIM assignments at time of report</td></tr>" }
 
$pimEligRows = ($pimEligData | ForEach-Object {
    "<tr><td>$($_.principal)</td><td>$($_.role)</td><td>$(TierBadge $_.tier)</td>" +
    "<td>$($_.maxDuration)</td><td>$($_.approval)</td><td>$($_.mfa)</td></tr>"
}) -join ""
 
$pimHistRows = ($pimHistData | ForEach-Object {
    $rowCls = if ($_.result -eq "failure") { "hist-row-fail" } else { "hist-row-ok" }
    $just   = if ($_.justification) { $_.justification } else { "<span style='color:#b71c1c'>No justification recorded</span>" }
    "<tr class='$rowCls' data-result='$($_.result)'>" +
    "<td>$($_.date)</td><td>$($_.principal)</td><td>$($_.role)</td>" +
    "<td style='font-size:11px'>$just</td>" +
    "<td style='font-size:11px'>$($_.approvedBy)</td><td>$(ResultBadge $_.result)</td></tr>"
}) -join ""
