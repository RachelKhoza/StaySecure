<#
.SYNOPSIS
    Checks a list of Azure VMs (by VM name only) for pending security updates,
    last patch info (date + time), and pending-reboot state. Read-only.
 
.DESCRIPTION
    Input is a CSV (or array) of VM names. The script:
        1. Connects to a subscription
        2. Enumerates all VMs in that subscription once
        3. Resolves each VM name to its resource group automatically
        4. Runs an in-guest probe via Invoke-AzVMRunCommand
        5. Reports per-VM:
             - PendingSecurityUpdates  (count + KB list)
             - LastPatchInstalledUtc   (timestamp of most recent hotfix)
             - LastPatchKB
             - RecentPatches           (last 5 hotfixes with timestamps)
             - RebootPending + reasons
             - PowerState
 
    Read-only. Does not install or change anything on the VMs.
 
.PARAMETER SubscriptionId
    Azure subscription to target. If omitted, uses current Az context.
 
.PARAMETER CsvPath
    Path to a CSV with a 'VmName' column. Header optional - if file is just a
    plain list of VM names (one per line), it works too.
 
.PARAMETER VmList
    Alternative to CsvPath: pass VM names directly.
 
.PARAMETER OutputCsv
    If set, writes the result table to this CSV path.
 
.PARAMETER ThrottleLimit
    If >1, runs probes in parallel (PowerShell 7+ required).
 
.EXAMPLE
    .\Check-VmPatchStatus.ps1 -CsvPath .\vms.csv
 
.EXAMPLE
    .\Check-VmPatchStatus.ps1 -CsvPath .\vms.csv -SubscriptionId "xxxx-xxxx" -OutputCsv .\report.csv
 
.EXAMPLE
    .\Check-VmPatchStatus.ps1 -VmList "EMEAWEUT00AZDE","EMEAWEUT0001"
#>
 
[CmdletBinding(DefaultParameterSetName = "Csv")]
param(
    [Parameter(ParameterSetName = "Csv", Mandatory = $true)]
    [string]$CsvPath,
 
    [Parameter(ParameterSetName = "List", Mandatory = $true)]
    [string[]]$VmList,
 
    [string]$SubscriptionId,
 
    [string]$OutputCsv,
 
    [int]$ThrottleLimit = 1
)
 
$ErrorActionPreference = "Stop"
 
# =========================================================
# Helpers
# =========================================================
function Write-Log {
    param([string]$Message, [string]$Level = "INFO")
    $color = switch ($Level) {
        "ERROR" { "Red" }
        "WARN"  { "Yellow" }
        "OK"    { "Green" }
        default { "Cyan" }
    }
    Write-Host "[$(Get-Date -Format 'HH:mm:ss')] [$Level] $Message" -ForegroundColor $color
}
 
function Read-VmNames {
    param(
        [string]$CsvPath,
        [string[]]$VmList
    )
 
    if ($VmList) {
        return $VmList | Where-Object { $_ } | ForEach-Object { $_.Trim() } | Sort-Object -Unique
    }
 
    if (-not (Test-Path $CsvPath)) {
        throw "CSV not found: $CsvPath"
    }
 
    $names = @()
 
    # Try as CSV with header first
    try {
        $rows = Import-Csv $CsvPath
        if ($rows -and ($rows[0].PSObject.Properties.Name -contains "VmName")) {
            $names = $rows | ForEach-Object { $_.VmName } | Where-Object { $_ }
        }
        elseif ($rows -and ($rows[0].PSObject.Properties.Name -contains "Name")) {
            $names = $rows | ForEach-Object { $_.Name } | Where-Object { $_ }
        }
    } catch { }
 
    # Fallback: treat file as plain list of names, one per line
    if (-not $names -or $names.Count -eq 0) {
        $names = Get-Content $CsvPath |
            Where-Object { $_ -and $_.Trim() -and $_.Trim() -notmatch "^(VmName|Name)$" } |
            ForEach-Object { $_.Trim() }
    }
 
    if (-not $names -or $names.Count -eq 0) {
        throw "No VM names found in $CsvPath. Expected a CSV with 'VmName' column or one VM name per line."
    }
 
    $names | ForEach-Object { $_.Trim() } | Sort-Object -Unique
}
 
function Ensure-AzContext {
    param([string]$SubscriptionId)
 
    $ctx = Get-AzContext -ErrorAction SilentlyContinue
    if (-not $ctx) {
        Write-Log "No Az context. Running Connect-AzAccount..." "WARN"
        Connect-AzAccount | Out-Null
        $ctx = Get-AzContext
    }
 
    if ($SubscriptionId -and $ctx.Subscription.Id -ne $SubscriptionId) {
        Write-Log "Switching to subscription $SubscriptionId" "INFO"
        Set-AzContext -SubscriptionId $SubscriptionId | Out-Null
        $ctx = Get-AzContext
    }
 
    Write-Log "Subscription: $($ctx.Subscription.Name) ($($ctx.Subscription.Id))" "OK"
    Write-Log "Account     : $($ctx.Account.Id)" "OK"
}
 
function Resolve-VmsByName {
    <#
        Enumerates ALL VMs in the current subscription once, then maps the
        requested names to resource groups. Returns:
          - resolved     : array of @{ Name; ResourceGroupName }
          - notFound     : array of names not present in subscription
          - duplicates   : array of names that appear more than once
    #>
    param([string[]]$VmNames)
 
    Write-Log "Enumerating VMs in subscription (one-time scan)..." "INFO"
    $all = Get-AzVM
    Write-Log "Found $($all.Count) VMs total in subscription." "OK"
 
    $resolved   = @()
    $notFound   = @()
    $duplicates = @()
 
    foreach ($name in $VmNames) {
        $vmMatches = $all | Where-Object { $_.Name -eq $name }
        if (-not $vmMatches) {
            $notFound += $name
            continue
        }
        if (@($vmMatches).Count -gt 1) {
            $duplicates += $name
            $resolved += [PSCustomObject]@{
                Name              = $name
                ResourceGroupName = $vmMatches[0].ResourceGroupName
                Ambiguous         = $true
                AllResourceGroups = ($vmMatches.ResourceGroupName -join "; ")
            }
        }
        else {
            $resolved += [PSCustomObject]@{
                Name              = $name
                ResourceGroupName = $vmMatches.ResourceGroupName
                Ambiguous         = $false
                AllResourceGroups = $vmMatches.ResourceGroupName
            }
        }
    }
 
    [PSCustomObject]@{
        Resolved   = $resolved
        NotFound   = $notFound
        Duplicates = $duplicates
    }
}
 
# In-guest probe script
$probeScript = @'
$ErrorActionPreference = "Continue"
$result = [ordered]@{
    HostName                  = $env:COMPUTERNAME
    PSWindowsUpdateAvailable  = $false
    PendingSecurityUpdates    = 0
    PendingSecurityKBs        = ""
    PendingAllUpdates         = 0
    LastPatchInstalledUtc     = ""
    LastPatchInstalledLocal   = ""
    LastPatchKB               = ""
    RecentPatches             = ""
    RebootPending             = $false
    RebootReasons             = ""
    OsCaption                 = ""
    OsVersion                 = ""
    ProbeError                = ""
}
 
try {
    # OS info (helps identify outliers)
    try {
        $os = Get-CimInstance Win32_OperatingSystem -ErrorAction Stop
        $result.OsCaption = $os.Caption
        $result.OsVersion = $os.Version
    } catch { }
 
    # ---- Patch history (date + time) ----
    # Strategy: combine Get-HotFix (fast) with WUA history (more accurate timestamps for some updates)
    $hotfixes = @()
    try {
        $hotfixes = Get-HotFix -ErrorAction Stop |
            Where-Object { $_.InstalledOn } |
            ForEach-Object {
                [PSCustomObject]@{
                    KB        = $_.HotFixID
                    Installed = $_.InstalledOn
                    Source    = "HotFix"
                    Title     = $_.Description
                }
            }
    } catch {
        $result.ProbeError += "HotFix:$($_.Exception.Message); "
    }
 
    # Update history via WUA (gives time-of-day, not just date)
    $wuaHistory = @()
    try {
        $session = New-Object -ComObject Microsoft.Update.Session
        $searcher = $session.CreateUpdateSearcher()
        $count = $searcher.GetTotalHistoryCount()
        if ($count -gt 0) {
            $hist = $searcher.QueryHistory(0, [Math]::Min($count, 200))
            foreach ($h in $hist) {
                # ResultCode 2 = Succeeded
                if ($h.ResultCode -eq 2 -and $h.Date) {
                    $kb = ""
                    if ($h.Title -match "(KB\d{6,7})") { $kb = $Matches[1] }
                    $wuaHistory += [PSCustomObject]@{
                        KB        = $kb
                        Installed = $h.Date
                        Source    = "WUA"
                        Title     = $h.Title
                    }
                }
            }
        }
    } catch {
        $result.ProbeError += "WUAHistory:$($_.Exception.Message); "
    }
 
    # Merge and pick the most recent
    $allPatches = @($hotfixes) + @($wuaHistory) |
        Where-Object { $_.Installed } |
        Sort-Object Installed -Descending
 
    if ($allPatches.Count -gt 0) {
        $latest = $allPatches[0]
        $result.LastPatchInstalledUtc   = $latest.Installed.ToUniversalTime().ToString("yyyy-MM-dd HH:mm:ss") + " UTC"
        $result.LastPatchInstalledLocal = $latest.Installed.ToString("yyyy-MM-dd HH:mm:ss")
        $result.LastPatchKB             = $latest.KB
 
        # Top 5 recent patches as a single semicolon-delimited field
        $top = $allPatches |
            Select-Object -First 5 |
            ForEach-Object {
                $kb = if ($_.KB) { $_.KB } else { "(no KB)" }
                "{0}@{1}" -f $kb, $_.Installed.ToString("yyyy-MM-dd HH:mm")
            }
        $result.RecentPatches = ($top -join "; ")
    }
 
    # ---- Reboot pending checks ----
    $rebootReasons = @()
    if (Test-Path "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\WindowsUpdate\Auto Update\RebootRequired") {
        $rebootReasons += "WindowsUpdate"
    }
    if (Test-Path "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Component Based Servicing\RebootPending") {
        $rebootReasons += "CBS"
    }
    try {
        $pfro = Get-ItemProperty "HKLM:\SYSTEM\CurrentControlSet\Control\Session Manager" -Name PendingFileRenameOperations -ErrorAction Stop
        if ($pfro.PendingFileRenameOperations) { $rebootReasons += "PendingFileRename" }
    } catch { }
    try {
        $ccm = Invoke-CimMethod -Namespace "root\ccm\ClientSDK" -ClassName CCM_ClientUtilities -MethodName DetermineIfRebootPending -ErrorAction Stop
        if ($ccm.RebootPending -or $ccm.IsHardRebootPending) { $rebootReasons += "SCCM" }
    } catch { }
 
    if ($rebootReasons.Count -gt 0) {
        $result.RebootPending = $true
        $result.RebootReasons = ($rebootReasons -join ",")
    }
 
    # ---- Pending updates scan ----
    $haveModule = [bool](Get-Module -ListAvailable -Name PSWindowsUpdate)
    $result.PSWindowsUpdateAvailable = $haveModule
 
    if ($haveModule) {
        try {
            Import-Module PSWindowsUpdate -ErrorAction Stop
            $allUpdates = Get-WindowsUpdate -MicrosoftUpdate -ErrorAction Stop
            $secUpdates = $allUpdates | Where-Object {
                $_.Categories -match "Security" -or $_.MsrcSeverity
            }
            $result.PendingAllUpdates      = @($allUpdates).Count
            $result.PendingSecurityUpdates = @($secUpdates).Count
            if ($secUpdates) {
                $kbs = $secUpdates | ForEach-Object {
                    if ($_.KB) { $_.KB }
                    elseif ($_.KBArticleIDs) { "KB" + ($_.KBArticleIDs -join ",KB") }
                    else { $_.Title }
                }
                $result.PendingSecurityKBs = ($kbs | Select-Object -Unique) -join ";"
            }
        } catch {
            $result.ProbeError += "PSWU:$($_.Exception.Message); "
        }
    }
    else {
        # Fallback: COM-based scan (no module install needed)
        try {
            $session2 = New-Object -ComObject Microsoft.Update.Session
            $searcher2 = $session2.CreateUpdateSearcher()
            $search   = $searcher2.Search("IsInstalled=0 and Type='Software' and IsHidden=0")
            $allUpdates = @($search.Updates)
            $secUpdates = $allUpdates | Where-Object {
                $_.Categories | Where-Object { $_.Name -match "Security" }
            }
            $result.PendingAllUpdates      = $allUpdates.Count
            $result.PendingSecurityUpdates = @($secUpdates).Count
            if ($secUpdates) {
                $kbs = foreach ($u in $secUpdates) {
                    if ($u.KBArticleIDs.Count -gt 0) {
                        foreach ($k in $u.KBArticleIDs) { "KB$k" }
                    } else { $u.Title }
                }
                $result.PendingSecurityKBs = ($kbs | Select-Object -Unique) -join ";"
            }
        } catch {
            $result.ProbeError += "COMSearch:$($_.Exception.Message); "
        }
    }
}
catch {
    $result.ProbeError += "Outer:$($_.Exception.Message)"
}
 
$json = $result | ConvertTo-Json -Compress
Write-Output "PROBE_RESULT_BEGIN"
Write-Output $json
Write-Output "PROBE_RESULT_END"
'@
 
function Invoke-VmProbe {
    param(
        [string]$ResourceGroupName,
        [string]$VmName,
        [string]$ScriptString,
        [bool]$Ambiguous,
        [string]$AllResourceGroups
    )
 
    $row = [ordered]@{
        VmName                   = $VmName
        ResourceGroupName        = $ResourceGroupName
        Ambiguous                = $Ambiguous
        AllResourceGroups        = $AllResourceGroups
        PowerState               = ""
        HostName                 = ""
        OsCaption                = ""
        PendingSecurityUpdates   = $null
        PendingSecurityKBs       = ""
        PendingAllUpdates        = $null
        LastPatchInstalledUtc    = ""
        LastPatchInstalledLocal  = ""
        LastPatchKB              = ""
        RecentPatches            = ""
        RebootPending            = $null
        RebootReasons            = ""
        PSWindowsUpdateAvailable = $null
        ProbeStatus              = ""
        ProbeError               = ""
        CheckedAt                = (Get-Date).ToString("yyyy-MM-dd HH:mm:ss")
    }
 
    try {
        $vm = Get-AzVM -ResourceGroupName $ResourceGroupName -Name $VmName -Status -ErrorAction Stop
        $row.PowerState = ($vm.Statuses | Where-Object Code -like "PowerState/*").DisplayStatus
    }
    catch {
        $row.ProbeStatus = "VM_NOT_FOUND"
        $row.ProbeError  = $_.Exception.Message
        return [PSCustomObject]$row
    }
 
    if ($row.PowerState -ne "VM running") {
        $row.ProbeStatus = "VM_NOT_RUNNING"
        return [PSCustomObject]$row
    }
 
    try {
        $job = Invoke-AzVMRunCommand `
            -ResourceGroupName $ResourceGroupName `
            -VMName $VmName `
            -CommandId "RunPowerShellScript" `
            -ScriptString $ScriptString `
            -ErrorAction Stop
 
        $output = ($job.Value | ForEach-Object { $_.Message }) -join "`n"
 
        if ($output -match "PROBE_RESULT_BEGIN\s*\r?\n([\s\S]*?)\r?\nPROBE_RESULT_END") {
            $json   = $Matches[1].Trim()
            $parsed = $json | ConvertFrom-Json
 
            $row.HostName                 = $parsed.HostName
            $row.OsCaption                = $parsed.OsCaption
            $row.PendingSecurityUpdates   = $parsed.PendingSecurityUpdates
            $row.PendingSecurityKBs       = $parsed.PendingSecurityKBs
            $row.PendingAllUpdates        = $parsed.PendingAllUpdates
            $row.LastPatchInstalledUtc    = $parsed.LastPatchInstalledUtc
            $row.LastPatchInstalledLocal  = $parsed.LastPatchInstalledLocal
            $row.LastPatchKB              = $parsed.LastPatchKB
            $row.RecentPatches            = $parsed.RecentPatches
            $row.RebootPending            = [bool]$parsed.RebootPending
            $row.RebootReasons            = $parsed.RebootReasons
            $row.PSWindowsUpdateAvailable = [bool]$parsed.PSWindowsUpdateAvailable
            $row.ProbeError               = $parsed.ProbeError
            $row.ProbeStatus              = if ($parsed.ProbeError) { "OK_WITH_WARNINGS" } else { "OK" }
        }
        else {
            $row.ProbeStatus = "NO_PROBE_OUTPUT"
            $row.ProbeError  = "Probe markers not found. Raw: $($output.Substring(0, [Math]::Min(300, $output.Length)))"
        }
    }
    catch {
        $row.ProbeStatus = "PROBE_FAILED"
        $row.ProbeError  = $_.Exception.Message
    }
 
    [PSCustomObject]$row
}
 
# =========================================================
# Main
# =========================================================
Write-Log "VM Patch Status Check - starting" "INFO"
 
Ensure-AzContext -SubscriptionId $SubscriptionId
 
$names = Read-VmNames -CsvPath $CsvPath -VmList $VmList
Write-Log "Loaded $($names.Count) VM name(s) from input." "INFO"
 
$lookup = Resolve-VmsByName -VmNames $names
 
if ($lookup.NotFound.Count -gt 0) {
    Write-Log "These VM names were NOT found in the subscription:" "WARN"
    foreach ($n in $lookup.NotFound) { Write-Host "    - $n" -ForegroundColor Yellow }
}
if ($lookup.Duplicates.Count -gt 0) {
    Write-Log "These VM names exist in MORE THAN ONE resource group (using first match, flagged as Ambiguous):" "WARN"
    foreach ($n in $lookup.Duplicates) { Write-Host "    - $n" -ForegroundColor Yellow }
}
 
$targets = $lookup.Resolved
Write-Log "Probing $($targets.Count) VM(s)..." "INFO"
 
$results = @()
 
if ($ThrottleLimit -gt 1 -and $PSVersionTable.PSVersion.Major -ge 7) {
    Write-Log "Running probes in parallel (throttle=$ThrottleLimit)" "INFO"
    $results = $targets | ForEach-Object -Parallel {
        $rg     = $_.ResourceGroupName
        $vm     = $_.Name
        $amb    = $_.Ambiguous
        $allRg  = $_.AllResourceGroups
        $script = $using:probeScript
 
        $invokeFn = ${using:function:Invoke-VmProbe}
 
        Write-Host "[$(Get-Date -Format 'HH:mm:ss')] Probing $vm (in $rg) ..."
        & ([scriptblock]::Create($invokeFn)) `
            -ResourceGroupName $rg `
            -VmName $vm `
            -ScriptString $script `
            -Ambiguous $amb `
            -AllResourceGroups $allRg
    } -ThrottleLimit $ThrottleLimit
}
else {
    foreach ($t in $targets) {
        Write-Log "Probing $($t.Name) (in $($t.ResourceGroupName)) ..." "INFO"
        $r = Invoke-VmProbe `
            -ResourceGroupName $t.ResourceGroupName `
            -VmName $t.Name `
            -ScriptString $probeScript `
            -Ambiguous $t.Ambiguous `
            -AllResourceGroups $t.AllResourceGroups
 
        $statusLevel = switch ($r.ProbeStatus) {
            "OK"               { "OK" }
            "OK_WITH_WARNINGS" { "WARN" }
            default            { "ERROR" }
        }
        Write-Log ("  -> {0} | PendingSec={1} | Reboot={2} | LastPatch={3}" -f `
            $r.ProbeStatus, $r.PendingSecurityUpdates, $r.RebootPending, $r.LastPatchInstalledLocal) $statusLevel
 
        $results += $r
    }
}
 
# Add synthetic rows for not-found VMs so the report is complete
foreach ($n in $lookup.NotFound) {
    $results += [PSCustomObject]@{
        VmName                   = $n
        ResourceGroupName        = ""
        Ambiguous                = $false
        AllResourceGroups        = ""
        PowerState               = ""
        HostName                 = ""
        OsCaption                = ""
        PendingSecurityUpdates   = $null
        PendingSecurityKBs       = ""
        PendingAllUpdates        = $null
        LastPatchInstalledUtc    = ""
        LastPatchInstalledLocal  = ""
        LastPatchKB              = ""
        RecentPatches            = ""
        RebootPending            = $null
        RebootReasons            = ""
        PSWindowsUpdateAvailable = $null
        ProbeStatus              = "VM_NOT_IN_SUBSCRIPTION"
        ProbeError               = "VM name not found in current subscription."
        CheckedAt                = (Get-Date).ToString("yyyy-MM-dd HH:mm:ss")
    }
}
 
# =========================================================
# Output
# =========================================================
Write-Host ""
Write-Host "================ PATCH STATUS REPORT ================" -ForegroundColor Cyan
 
$results |
    Select-Object VmName, ResourceGroupName, PowerState,
                  PendingSecurityUpdates, RebootPending,
                  LastPatchInstalledLocal, LastPatchKB,
                  ProbeStatus |
    Format-Table -AutoSize | Out-Host
 
# Detail block for VMs that need attention
$needsAttention = $results | Where-Object {
    $_.PendingSecurityUpdates -gt 0 -or
    $_.RebootPending -eq $true -or
    $_.ProbeStatus -notin @("OK", "OK_WITH_WARNINGS")
}
 
if ($needsAttention) {
    Write-Host ""
    Write-Host "----- VMs needing attention -----" -ForegroundColor Yellow
    foreach ($v in $needsAttention) {
        Write-Host ""
        Write-Host "$($v.VmName) ($($v.ResourceGroupName))" -ForegroundColor Yellow
        Write-Host "  Status         : $($v.ProbeStatus)"
        Write-Host "  PowerState     : $($v.PowerState)"
        if ($v.OsCaption)        { Write-Host "  OS             : $($v.OsCaption)" }
        Write-Host "  PendingSec     : $($v.PendingSecurityUpdates)"
        if ($v.PendingSecurityKBs) {
            Write-Host "  Pending KBs    : $($v.PendingSecurityKBs)"
        }
        Write-Host "  RebootPending  : $($v.RebootPending) ($($v.RebootReasons))"
        Write-Host "  LastPatchLocal : $($v.LastPatchInstalledLocal) ($($v.LastPatchKB))"
        if ($v.LastPatchInstalledUtc) {
            Write-Host "  LastPatchUtc   : $($v.LastPatchInstalledUtc)"
        }
        if ($v.RecentPatches) {
            Write-Host "  RecentPatches  : $($v.RecentPatches)"
        }
        if ($v.Ambiguous) {
            Write-Host "  Ambiguous      : True (also in: $($v.AllResourceGroups))" -ForegroundColor Magenta
        }
        if ($v.ProbeError) {
            Write-Host "  Error          : $($v.ProbeError)" -ForegroundColor Red
        }
    }
}
 
Write-Host ""
$summary = [ordered]@{
    Total                  = $results.Count
    Probed_OK              = ($results | Where-Object ProbeStatus -in "OK","OK_WITH_WARNINGS").Count
    NotInSubscription      = ($results | Where-Object ProbeStatus -eq "VM_NOT_IN_SUBSCRIPTION").Count
    NotRunning             = ($results | Where-Object ProbeStatus -eq "VM_NOT_RUNNING").Count
    ProbeFailed            = ($results | Where-Object ProbeStatus -eq "PROBE_FAILED").Count
    AmbiguousNames         = ($results | Where-Object Ambiguous -eq $true).Count
    WithPendingSecurity    = ($results | Where-Object { $_.PendingSecurityUpdates -gt 0 }).Count
    WithRebootPending      = ($results | Where-Object { $_.RebootPending -eq $true }).Count
}
Write-Host "Summary:" -ForegroundColor Cyan
$summary.GetEnumerator() | ForEach-Object { "  {0,-22}: {1}" -f $_.Key, $_.Value } | Write-Host
 
if ($OutputCsv) {
    $results | Export-Csv -Path $OutputCsv -NoTypeInformation -Force
    Write-Host ""
    Write-Log "Results exported to: $OutputCsv" "OK"
}
 
Write-Host ""
